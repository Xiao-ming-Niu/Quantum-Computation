\section{引言: 量子算法原语}

在上一模块中我们建立了量子计算的语言: \textbf{量子比特(qubit)},\textbf{酉门(unitary gate)},\textbf{电路(quantum circuit)}与\textbf{测量(measurement)}. 本模块介绍四类最基础的
\textbf{量子算法原语(quantum primitives)}: \textbf{量子傅里叶变换(quantum Fourier transform, QFT)},\textbf{量子相位估计(quantum phase estimation, QPE)},
\textbf{振幅放大(amplitude amplification, AA)}与\textbf{哈密顿量模拟(Hamiltonian simulation, 含 Lie--Trotter 公式(Lie--Trotter formula))}.
它们之所以被称为"构件", 是因为几乎所有量子算法都以它们为子程序:
QFT 是 QPE 的核心组成部分; QPE 是 \textbf{Shor 分解(Shor's algorithm)},\textbf{HHL 线性方程组(Harrow--Hassidim--Lloyd)},\textbf{本征值问题(eigenvalue problem)},\textbf{振幅估计(amplitude estimation)}的共同基础;
振幅放大在一切需要"放大成功概率"的场景中被反复调用; 哈密顿量模拟则通常作为 QPE 中酉算子
$U = e^{-iHt}$ 的实现方式被调用. 这些构件之间相互嵌套的关系构成了现代量子算法的骨架.

\begin{Note}
    按照 Lin Lin 讲义的划分, 上述四者(连同 \textbf{Grover 搜索(Grover's search)})构成最没有争议的量子原语集合.
    本模块先介绍不依赖其它构件的 QFT 与 AA, 再介绍把它们组合起来的 QPE 与 Trotter 型
    哈密顿量模拟. 更现代的构件(block-encoding, qubitization, QSVT)将在下一模块中
    统一并推广这些方法.
\end{Note}

\section{量子傅里叶变换 (QFT)}

\subsection{定义与基本性质}

\begin{Def}[QFT 的定义]
    设 $N = 2^n$. \textbf{量子傅里叶变换(quantum Fourier transform, QFT)}是
    $\mathbb{C}^N$ 上的酉算子 $U_{\mathrm{FT}}$, 作用在\textbf{计算基态(computational basis state)}上为
    \begin{equation}
        U_{\mathrm{FT}} \ket{j} = \frac{1}{\sqrt{N}} \sum_{k \in [N]} e^{2\pi i\, jk/N}\, \ket{k},
        \qquad j \in [N].
    \end{equation}
    其逆变换为
    \begin{equation}
        U_{\mathrm{FT}}^\dagger \ket{j} = \frac{1}{\sqrt{N}} \sum_{k \in [N]} e^{-2\pi i\, jk/N}\, \ket{k}.
    \end{equation}
\end{Def}

\begin{Prop}[基本性质]
    \begin{enumerate}
        \item $U_{\mathrm{FT}}$ 是酉算子($U_{\mathrm{FT}}^\dagger U_{\mathrm{FT}} = I$);
        \item $U_{\mathrm{FT}} \ket{0^n} = \frac{1}{\sqrt{N}} \sum_{k} \ket{k} = H^{\otimes n}\ket{0^n}$,
        即 QFT 把全零态变为均匀\textbf{叠加态(superposition state)};
        \item 与经典 \textbf{FFT(fast Fourier transform)} 的关键区别: QFT 的加速是指数级的(经典 FFT 需要 $O(N\log N)$ 次运算, 
        而 QFT 只需要 $O(n^2) = O(\log^2 N)$ 个门), 但 QFT 的\textbf{输出是叠加态},
        测量后只能得到单个基态, 无法直接读出全部傅里叶系数 --- 必须通过后续算法
        (如相位估计)来利用这一变换.
    \end{enumerate}
\end{Prop}

\begin{Note}[Hadamard 变换: $H^{\otimes n}$ 的作用]
    单比特 Hadamard 门 $H = \frac{1}{\sqrt{2}}\begin{pmatrix} 1 & 1 \\ 1 & -1 \end{pmatrix}$ 的作用为
    \begin{equation}
        H\ket{x} = \frac{1}{\sqrt{2}}\big(\ket{0} + (-1)^x\ket{1}\big)
        = \frac{1}{\sqrt{2}}\sum_{y \in \{0,1\}} (-1)^{xy}\,\ket{y}, \qquad x \in \{0,1\}.
    \end{equation}
    对 $n$ 比特施加 $H^{\otimes n} = H \otimes \cdots \otimes H$, 各比特独立作用, 
    逐个代入 $H\ket{x_i} = \frac{1}{\sqrt{2}}\sum_{y_i}(-1)^{x_i y_i}\ket{y_i}$:
    \begin{equation}
        H^{\otimes n}\ket{x_1\cdots x_n}
        = \frac{1}{\sqrt{2^n}}\sum_{y_1,\dots,y_n \in \{0,1\}}
           (-1)^{x_1 y_1 + \cdots + x_n y_n}\,\ket{y_1\cdots y_n}
        = \frac{1}{\sqrt{2^n}}\sum_{y \in \{0,1\}^n} (-1)^{x\cdot y}\,\ket{y},
    \end{equation}
    其中 $x \cdot y := \sum_i x_i y_i \bmod 2$ 是二进制向量的\textbf{模 2 内积}.
    这就是\textbf{Walsh--Hadamard 变换}. 两个重要特例:
    \begin{enumerate}
        \item \textbf{作用于 $\ket{0^n}$}: 因 $0 \cdot y = 0$ 相位恒为 $+1$, 得到\textbf{均匀叠加态}
        \[ H^{\otimes n}\ket{0^n} = \frac{1}{\sqrt{2^n}}\sum_{y\in\{0,1\}^n}\ket{y}; \]
        \item \textbf{作用于 $\ket{1^n}$}: 因 $1^n \cdot y = |y|$($y$ 中 $1$ 的个数), 得到带交替相位的叠加
        \[ H^{\otimes n}\ket{1^n} = \frac{1}{\sqrt{2^n}}\sum_{y\in\{0,1\}^n} (-1)^{|y|}\,\ket{y}. \]
    \end{enumerate}
    均匀叠加态 $H^{\otimes n}\ket{0^n}$ 是 Grover 搜索的初态, 也是 QPE 中相位寄存器
    $H^{\otimes t}\ket{0^t}$ 的制备方式; QFT 正是把 Hadamard 变换推广到含相位旋转 $\omega^{jk}$ 的变换.
\end{Note}

\subsection{二进制分解与张量积结构}

把 $j, k$ 写成二进制 $j = (j_{n-1}\cdots j_0)$, $k = (k_{n-1}\cdots k_0)$, 则

\begin{equation}
    \frac{jk}{N} = k_0\, (0.j_{n-1}\cdots j_0) + k_1\, (0.j_{n-2}\cdots j_0) + \cdots + k_{n-1}\, (0.j_0),
\end{equation}

其中 $0.j_{m}\cdots j_0$ 表示二进制小数 $j_m/2 + \cdots + j_0/2^{m+1}$. 于是指数因子可以分解, 
代入定义并逐比特求和, 得到 QFT 的\textbf{张量积(tensor product)}形式:

\begin{equation}
    U_{\mathrm{FT}} \ket{j_{n-1}\cdots j_0}
    = \frac{1}{\sqrt{2^n}} \Big(\ket{0} + e^{2\pi i (0.j_0)}\ket{1}\Big)
    \Big(\ket{0} + e^{2\pi i (0.j_1 j_0)}\ket{1}\Big)
    \cdots \Big(\ket{0} + e^{2\pi i (0.j_{n-1}\cdots j_0)}\ket{1}\Big).
\end{equation}

\begin{Proof}[张量积形式的推导]
    把二进制展开式
    \[ j = j_{n-1}2^{n-1}+j_{n-2}2^{n-2}+\cdots+j_1 2^1+j_0 2^0 \]
    代入定义, 指数因子按 $k_{n-1}, \dots, k_0$ 分离:
    \begin{equation*}
    \begin{aligned}
        U_{\mathrm{FT}} \ket{j_{n-1}\cdots j_0}
        &= \frac{1}{\sqrt{2^n}}\sum_{k_{n-1},\dots,k_0}
           e^{2\pi i\big(k_0(0.j_{n-1}\cdots j_0) + \cdots + k_{n-1}(0.j_0)\big)}
           \ket{k_{n-1}\cdots k_0} \\
        &= \frac{1}{\sqrt{2^n}}
           \Big(\sum_{k_{n-1}} e^{2\pi i k_{n-1}(0.j_0)}\ket{k_{n-1}}\Big)
           \otimes \cdots \otimes
           \Big(\sum_{k_0} e^{2\pi i k_0(0.j_{n-1}\cdots j_0)}\ket{k_0}\Big).
    \end{aligned}
    \end{equation*}
    第二步把 $n$ 重求和写成张量积, 是因为第 $m$ 个指数因子只依赖对应的求和指标
    $k_m$(与其它 $k$ 无关). 每个因子都是一个单比特叠加
    \[ \sum_{k\in\{0,1\}} e^{2\pi i k\varphi}\ket{k} = \ket{0} + e^{2\pi i\varphi}\ket{1}, \]
    代入即得上式. 注意张量积的因子顺序与输出比特 $k_{n-1}, \dots, k_0$ 一致.
\end{Proof}

注意: 第一个因子(对应 $\ket{k_{n-1}}$)只依赖最低比特 $j_0$, 而最后一个因子
(对应 $\ket{k_0}$)依赖全部比特 --- 即\textbf{输出比特的位序与输入相反}.

\subsection{电路实现}

实现的关键是"\textbf{受控相位旋转(controlled phase rotation)}". 记单比特旋转门
\begin{equation}
    R_z(\varphi) = \begin{pmatrix} 1 & 0 \\ 0 & e^{i\varphi} \end{pmatrix},
    \qquad R_m := R_z(2\pi/2^{m}).
\end{equation}

\begin{Eg}[受控旋转的实现]
    目标: 把 $j_{n-1}, \dots, j_0$ 的相位信息收集到辅助比特上:
    \[ \ket{0}\ket{j} \;\longrightarrow\; \frac{1}{\sqrt{2}}\Big(\ket{0} + e^{2\pi i(0.j_{n-1}\cdots j_0)}\ket{1}\Big)\ket{j}. \]
    利用
    \[ e^{2\pi i (0.j_{n-1}\cdots j_0)} = e^{2\pi i (0.j_{n-1})}\, e^{2\pi i (0.0 j_{n-2})} \cdots e^{2\pi i (0.\underbrace{0\cdots 0}_{n-2}\, j_0)}, \]
    该操作等价于依次施加 $H$ 与以 $j_{m}$ 为控制比特的受控 $R_{m+1}$ 门
    ($m = 0, \dots, n-1$): 每个比特 $j_m$ 贡献相位 $e^{2\pi i (0.0\cdots 0 j_m)}$.
\end{Eg}

QFT 电路(以 3 比特为例, 输出位序已反转, 需用 SWAP 门恢复):

\begin{equation*}
\Qcircuit @C=.9em @R=.8em {
    \lstick{\ket{j_2}} & \gate{H} & \gate{R_2} & \gate{R_3} & \qw & \qw & \qw & \qw & \qw & \qw & \qw & \rstick{\ket{k_0}} \qw \\
    \lstick{\ket{j_1}} & \qw & \ctrl{-1} & \qw & \gate{H} & \gate{R_2} & \qw & \qw & \qw & \qw & \qw & \rstick{\ket{k_1}} \qw \\
    \lstick{\ket{j_0}} & \qw & \qw & \ctrl{-2} & \qw & \ctrl{-1} & \gate{H} & \qw & \qw & \qw & \qw & \rstick{\ket{k_2}} \qw
}
\end{equation*}

\begin{Thm}[QFT 的复杂度]
    $n$ 比特 QFT 可以用 $O(n^2)$ 个单比特与\textbf{受控门(controlled gate)}实现, 电路深度为 $O(n)$.
    逆变换 $U_{\mathrm{FT}}^\dagger$ 只需把所有受控旋转替换为共轭 $R_m^\dagger$ 并反向执行.
\end{Thm}

\begin{Proof}
    设比特 $j_0, \dots, j_{n-1}$ 自下而上排列. 比特 $j_m$($m = 0, \dots, n-1$)
    上需要 1 个 $H$ 门与 $m$ 个受控旋转 $R_2, \dots, R_{m+1}$
    (分别以更低位为控制), 共 $\sum_{m=0}^{n-1}(m+1) = \frac{n(n+1)}{2}$ 个基本门;
    恢复位序还需 $\lfloor n/2 \rfloor$ 个 SWAP 门, 因此总门数为 $O(n^2)$.
    由于同列的门作用在不同的量子比特上, 可以并行, 从输入到输出的最长路径
    只经过 $O(n)$ 个门, 电路深度为 $O(n)$. 逆变换的论证完全相同:
    受控旋转取共轭 $R_m^\dagger$(即旋转角变号), 位序恢复仍用 SWAP.
\end{Proof}

\subsection{与经典 DFT/FFT 的对比}

经典计算中的\textbf{离散傅里叶变换(discrete Fourier transform, DFT)}把长度为 $N$
的向量 $\{a_j\}_{j=0}^{N-1}$ 映射为
\begin{equation}
    \hat{a}_k = \sum_{j=0}^{N-1} e^{-2\pi i jk/N}\, a_j, \qquad k = 0, \dots, N-1,
\end{equation}
直接计算需要 $O(N^2)$ 次运算; \textbf{快速傅里叶变换(fast Fourier transform, FFT)},
即 Cooley--Tukey 分治算法, 把复杂度降到 $O(N\log N)$, 是经典科学计算(信号处理, 
\textbf{谱方法(spectral method)}解 PDE, 卷积)的基石. QFT 与二者的对比见表 \ref{tab:ft}:

\begin{table}[H]
\centering
\begin{tabular}{lccc}
    \toprule
    方法 & 输入 & 复杂度 & 输出 \\
    \midrule
    DFT(直接求和) & 经典向量 $\{a_j\}$ & $O(N^2)$ & 经典向量(可直接读出) \\
    FFT(分治) & 经典向量 $\{a_j\}$ & $O(N\log N)$ & 经典向量(可直接读出) \\
    QFT(量子线路) & 量子态 $\ket{\psi}=\sum_j a_j\ket{j}$ & $O(\log^2 N)$ 个门 & 量子叠加态 \\
    \bottomrule
\end{tabular}
\caption{DFT, FFT 与 QFT 的对比. 门复杂度上 QFT 对 FFT 是指数级加速, 
但两者处理的对象与输出本质不同.}
\label{tab:ft}
\end{table}

\begin{Note}[为什么 QFT 不能直接替代 FFT]
    QFT 作用在\textbf{振幅(amplitude)}上而非数据向量上, 且输出是量子叠加态:
    测量一次只能得到一个基态, 读出全部 $N$ 个系数需要 $\Omega(N)$ 次测量
    (且每个系数还需多次重复采样才能估计到精度). 因此 QFT 的优势不在于
    "更快地计算经典傅里叶变换", 而在于它能作为\textbf{子程序}嵌入更大规模的
    量子算法(QPE, Shor 分解, 量子线性代数等): 在这些算法中, "傅里叶变换"
    作用在纠缠的振幅上, 中间结果无需读出, 只有最终观测才消耗测量.
    这与经典\textbf{谱方法}中的角色类似但相反: 经典谱方法用 FFT 在
    $O(N\log N)$ 内完成正逆变换, 读出谱系数后显式组装解; 量子对应物
    (QFT 或其对角化 block-encoding)则把变换"藏在"线路内部, 
    避免读出的指数代价 --- 代价是只能访问振幅(叠加)上的结果.
\end{Note}

\begin{Note}[Shor 算法中的角色]
    Shor 的整数分解算法正是"QFT 无需读出"这一性质的极致体现:
    周期 $r$ 的信息被编码在相位估计的振幅中, 逆 QFT 后\textbf{一次测量}
    即可(高概率)读出与 $r$ 相关的相位 --- 经典计算中同样的周期搜索
    需要做 $\Omega(N)$ 次乘幂.
\end{Note}

\section{量子相位估计 (QPE)}

\subsection{问题设定与 Hadamard 检验}

\begin{Def}[相位估计问题]
    设 $U$ 是酉算子, $\ket{\psi}$ 是其特征向量:
    \begin{equation}
        U \ket{\psi} = e^{2\pi i \theta} \ket{\psi}, \qquad \theta \in [0, 1).
    \end{equation}
    给定 \textbf{oracle} 访问 $U$(及其幂), 目标是估计 $\theta$ 到一定精度.
\end{Def}

\begin{Note}
    相位估计是最重要的量子原语之一: Shor 分解, HHL 线性方程组, 本征值问题, 振幅估计, 
    \textbf{量子行走(quantum walk)}等都以它为子程序.
\end{Note}

最简单的相位估计是 \textbf{Hadamard 检验(Hadamard test)}(已在 Module1 中介绍): 对控制比特先施加
$H$, 作用受控 $U$ 后再施加 $H$, 则测量结果为 $1$ 的概率满足

\begin{equation}
    p(1) = \frac{1}{2}\big(1 - \cos(2\pi\theta)\big),
\end{equation}

从而 $\theta = \frac{1}{2\pi}\arccos\big(1 - 2p(1)\big) \pmod 1$.
但该方法的精度受限于采样误差: 由 Module1 的分析, 达到加性精度 $\varepsilon$
需要 $O(1/\varepsilon^2)$ 次重复测量. 相位估计的任务正是把精度提高到 $O(1/\varepsilon)$.

\begin{Note}[与经典本征值/相位估计的对比]
    经典计算中估计厄米矩阵的本征值或酉算子的本征相位, 常用\textbf{幂迭代(power iteration)},
    QR 算法或 \textbf{Krylov 子空间(Krylov subspace)}方法(如 Lanczos): 幂迭代/Lanczos 的迭代次数对\textbf{谱间隙(spectral gap)}的依赖为
    $O(1/\Delta)$ 或 $O(1/\sqrt{\Delta})$, 且每步需要 $O(N^2)$--$O(N^3)$ 次矩阵运算
    (或 $O(\mathrm{nnz})$ 次稀疏运算), 并\textbf{显式存储} $N \times N$ 矩阵.
    QPE 与之有三点本质区别:
    \begin{enumerate}
        \item QPE 通过\textbf{黑盒 oracle(black-box oracle)}访问 $U^{2^j}$, 无需显式构造矩阵
        (代价是 $O(1/\varepsilon)$ 次\textbf{查询(query)}, 其中 $\varepsilon$ 为相位精度);
        \item QPE 对 $N = 2^n$ 的依赖是 $\mathrm{poly}(n) = \mathrm{poly}(\log N)$,
        经典方法至少线性于 $N$;
        \item QPE 可同时作用于本征态的\textbf{叠加}, 一次运行即可读出全部本征相位
        (概率由振幅 $|c_j|^2$ 给出) --- 经典 Krylov 方法同样可以, 但每步代价更高.
    \end{enumerate}
    这些差异是 HHL, Shor 等算法能够工作的根本原因: 它们把"求解"建立在
    oracle 查询之上, 而不是对显式矩阵做数值线性代数.
\end{Note}

\subsection{受控 $U^{2^j}$ 与二进制分解}

把相位 $\theta = (0.\theta_{d-1}\cdots\theta_0)$ 写成 $d$ 位二进制小数, 考虑一个"大"\textbf{受控酉算子(controlled-unitary operator)}

\begin{equation}
    \mathcal{U} := \sum_{j \in [2^d]} \ket{j}\bra{j} \otimes U^j.
\end{equation}

看似需要实现全部 $2^d$ 个不同的 $U^j$, 但利用二进制展开
$j = \sum_i j_i 2^i$ 得 $U^j = \prod_i U^{j_i 2^i}$, 于是

\begin{equation}
    \begin{aligned}
        \mathcal{U} &= \sum_{j \in [2^d]} \ket{j}\bra{j} \otimes \prod_i U^{j_i 2^i} \\
        &= \sum_{j_{d-1},\cdots,j_0}(\ket{j_{d-1}}\bra{j_{d-1}} \otimes \cdots \otimes \ket{j_0}\bra{j_0}) \otimes \prod_i U^{j_i 2^i} \\
        &= \sideset{}{'}\prod_{i=0}^{d-1} \Big(\sum_{j_i}\ket{j_i}\bra{j_i}\otimes U^{j_i 2^i}\Big) \\
        &= \sideset{}{'}\prod_{i=0}^{d-1} \Big(\ket{0}\bra{0} \otimes I + \ket{1}\bra{1} \otimes U^{2^i}\Big),
    \end{aligned}
\end{equation}

即只需 $d$ 个受控 $U^{2^i}$ 门. 若 $U$ 可以被 \textbf{快速前推(fast-forwarded)}
(实现 $U^j$ 的代价与 $j$ 无关, 例如 $U = R_z(\varphi)$), 该分解比直接实现 $\mathcal{U}$
指数级节省资源.

\subsection{QFT 基 QPE 电路}

设 $T = 2^t$, 其中 $t$ 为\textbf{相位寄存器(phase register)}比特数. QPE 的电路结构为:
\begin{equation*}
\Qcircuit @C=1em @R=.8em {
    \lstick{\ket{0}^{\otimes t}} & \gate{H^{\otimes t}} & \multigate{1}{\mathcal{U}} & \gate{U_{\mathrm{FT}}^\dagger} & \meter & \rstick{\tilde{\theta}} \qw \\
    \lstick{\ket{\psi}} & \qw & \ghost{\mathcal{U}} & \qw & \qw & \qw
}
\end{equation*}

推导如下:

\begin{equation}
    \ket{0^t}\ket{\psi} \;\xrightarrow{H^{\otimes t}}\;
    \frac{1}{\sqrt{T}}\sum_{j\in[T]} \ket{j}\ket{\psi}
    \;\xrightarrow{\mathcal{U}}\;
    \frac{1}{\sqrt{T}}\sum_{j\in[T]} e^{2\pi i j\theta} \ket{j}\ket{\psi}
    \;\xrightarrow{U_{\mathrm{FT}}^\dagger}\;
    \sum_{k\in[T]} \alpha_k \ket{k}\ket{\psi},
\end{equation}

其中

\begin{equation}
    \alpha_k = \frac{1}{T}\sum_{j\in[T]} e^{2\pi i j(\theta - k/T)}
    = \frac{1}{T}\cdot\frac{1 - e^{2\pi i T(\theta - k/T)}}{1 - e^{2\pi i (\theta - k/T)}}.
\end{equation}

因此有
\begin{equation}
    Pr(k) = |\alpha_k|^2 = \frac{1}{T^2}\,\frac{\sin^2(\pi T(\theta - k/T))}{\sin^2(\pi (\theta - k/T))}.
\end{equation}

\begin{Thm}[精确相位的确定性]
    若 $\theta = k_0/T$ 恰有 $t$ 位二进制表示(即 $\theta = (0.\theta_{t-1}\cdots\theta_0)$),
    则 $\alpha_k = \delta_{k,k_0}$, 测量相位寄存器\textbf{确定性地}得到 $k_0$.
\end{Thm}

\begin{Eg}[d = 1 的特殊情形]
    当 $t = 1$ 时 $U_{\mathrm{FT}} = H$, 上述电路退化为 Hadamard 检验.
    这说明 Hadamard 检验正是 QPE 在 1 比特相位寄存器下的特例;
    且此时 $\theta$ 不必是单比特数 --- 一般情形由下述误差分析处理.
\end{Eg}

\subsection{误差分析}

当 $\theta_0$ 不能由 $t$ 位二进制精确表示时, 测量结果是随机的.
记周期距离 $\norm{x}_1 := \min\{x \bmod 1,\ 1 - (x \bmod 1)\}$, 由几何级数公式
$|1 - e^{i\varphi}| \ge 2|\varphi|/\pi$ 可得

\begin{equation}
    |\alpha_k| \le \frac{1}{2T\, \norm{\theta_0 - k/T}_1}.
\end{equation}

\begin{Thm}[QPE 的精度与成功概率]
    设精度参数 $\varepsilon = 2^{-d}$.
    (1) \textbf{常数成功概率}: 取相位寄存器比特数 $t = d + O(1)$(例如 $t = d + 2$),
    则 $T = 2^t = O(1/\varepsilon)$, 测量结果 $k_0$ 以\textbf{常数}概率(如 $\ge 3/4$)
    满足 $\norm{\theta_0 - k_0/T}_1 \le \varepsilon$;
    (2) \textbf{高成功概率}: 若要求一次运行的失败概率 $\le \varepsilon$, 取
    \begin{equation}
        t = d + \lceil \log_2(\varepsilon^{-1}) \rceil, \qquad
        T = 2^t = O(\varepsilon^{-2}),
    \end{equation}
    此时 $P\Big(\norm{\theta_0 - k_0/T}_1 \le \varepsilon\Big) \ge 1 - \varepsilon$.
    两种情形下, QPE 的查询复杂度 $O(T)$ 都满足: 在常数成功概率下
    $O(T) = O(1/\varepsilon)$, 比 Hadamard 检验的 $O(1/\varepsilon^2)$ 是\textbf{平方级改进}.
\end{Thm}

\begin{Proof}
    (1) 单个振幅的界: 记 $\delta_k := \theta_0 - k/T$. 由几何级数求和公式
    $\sum_{j\in[T]} q^j = (1-q^T)/(1-q)$ 与不等式 $|1 - e^{i\varphi}| \ge 2|\varphi|/\pi$,
    \begin{equation}
        |\alpha_k| = \Big|\frac{1 - e^{2\pi i T\delta_k}}{T(1 - e^{2\pi i \delta_k})}\Big|
        \le \frac{1}{2T\, \norm{\theta_0 - k/T}_1}.
    \end{equation}
    (2) 失败概率的界: 由于 $k/T$ 在单位圆周上等距分布(间距 $1/T$), 用积分放缩
    (距离 $\theta_0$ 超过 $\varepsilon$ 的项求和, 两侧各留一个距离约 $1/2$ 的边界项):
    \begin{equation}
        P\Big(\norm{\theta_0 - k_0/T}_1 \ge \varepsilon\Big)
        \le \sum_{\norm{\theta_0 - k/T}_1 \ge \varepsilon}
        \frac{1}{4T^2\, \norm{\theta_0 - k/T}_1^2}
        \le \frac{1}{2T\varepsilon} + \frac{1}{2(T\varepsilon)^2}.
    \end{equation}
    (3) 情形 (1): 取 $t = d + c$($c$ 常数), 则 $T\varepsilon = 2^c$, 失败概率
    $\le 2^{-c-1} + 2^{-2c-1}$; 取 $c = 2$ 得 $\le 1/8 + 1/32 = 5/32 < 1/4$;
    (4) 情形 (2): 取 $t = d + \lceil\log_2(\varepsilon^{-1})\rceil$, 则
    $T\varepsilon \ge \varepsilon^{-1}$, 失败概率 $\le \varepsilon/2 + \varepsilon^2/2 \le \varepsilon$.
    这与 Nielsen--Chuang 的结论 $t = n + \lceil\log_2(2 + 1/(2\delta))\rceil$ 一致
    ($n$ 为精度比特数, $\delta$ 为失败概率): 常数 $\delta$ 时 $t = n + O(1)$,
    $\delta = \varepsilon$ 时 $t = n + \log_2(1/\varepsilon)$.
\end{Proof}

图 \ref{fig:qpe} 给出了测量概率 $p(k) = |\alpha_k|^2$ 的分布形状: 在
$k_0 \approx T\theta_0$ 附近形成一个宽度约为 $1/T$ 的窄峰, 峰外概率迅速衰减
(这是"以 $1/\varepsilon$ 的模拟时间换取 $1/\varepsilon$ 的相位精度"的几何直观).

\begin{Note}[量子中位数法与两个假设的放松]
    上述分析默认 $\ket{\psi}$ 恰为特征向量且相位有精确二进制表示.
    (1) 若初态是特征向量的线性组合 $\ket{\psi} = \sum_k c_k \ket{k}$,
    则 QPE 以概率 $|c_0|^2$ 同时返回目标相位与本征态 --- 这正是基态能量估计和 HHL 的工作方式;
    (2) 为避免因 $\log_2(1/\varepsilon)$ 个额外比特增加电路深度, 可把 QPE 运行 $\log(1/\varepsilon)$ 次
    并取相位的中位数(\textbf{量子中位数法(quantum median)}), 以 $\log(1/\varepsilon)$ 次重复代替额外的比特数与模拟时间.
\end{Note}

\begin{Note}[Fejér 核与测量分布的闭式表达]
    测量概率 $P(k) = |\alpha_k|^2$ 可以写成闭式. 由几何级数求和与
    $|1 - e^{2\pi i x}| = 2|\sin(\pi x)|$:
    \begin{equation}
        P(k) = |\alpha_k|^2 = \frac{1}{T^2}\,\frac{\sin^2\!\big(\pi T(\theta - k/T)\big)}{\sin^2\!\big(\pi(\theta - k/T)\big)}
        =: F_T\!\Big(\theta - \frac{k}{T}\Big),
    \end{equation}
    其中
    \begin{equation}
        F_T(x) := \frac{1}{T^2}\,\frac{\sin^2(\pi T x)}{\sin^2(\pi x)}
    \end{equation}
    称为 \textbf{Fejér 核}(Fejér kernel). $F_T$ 在 $x = 0$ 处取最大值 $1$
    (由 $F_T(0) = \lim_{x\to 0}F_T(x) = 1$), 并在 $|x| \gtrsim 1/T$ 处迅速衰减 ---
    这正是"相位 $\theta$ 附近宽度约 $1/T$ 的窄峰"的精确分析来源(见图 \ref{fig:qpe}).
\end{Note}

\begin{Note}[近似实现 $U$ 的误差分离处理]
    QPE 分析中的理想化条件之一是 $U$ 可精确实现. 若只能用 $\tilde U$ 近似 $U$ 且
    $\norm{\tilde U - U} \le \varepsilon$, 则由酉矩阵的扰动理论, 单位圆上本征值的扰动至多为
    $\varepsilon$, 相应\textbf{相角}的扰动至多为 $\varepsilon/4\pi$. 因此\textbf{实现误差可以作为
    相角误差的加性项单独计入}, 其余分析仍可在 $U$ 精确实现的假设下进行. 实际中
    $U = e^{-iHt}$ 由 Hamiltonian 模拟近似实现, 其模拟误差正是这一项;
    HHL 等应用正是把 QPE 的相位误差与 Hamiltonian 模拟误差分开讨论.
\end{Note}

\begin{Prop}[一般初态(特征态叠加)的测量分布]
    设 $U\ket{k'} = e^{2\pi i\lambda_{k'}}\ket{k'}$ 为本征分解, 初态为叠加
    $\ket{\psi} = \sum_{k'} c_{k'}\ket{k'}$. 以 $t$ 比特 QPE 电路($T = 2^t$)作用于
    $\ket{0^t}\ket{\psi}$, 对相位寄存器测量得到 $k \in [T]$ 的概率为
    \begin{equation}
        P(k) = \sum_{k'} |c_{k'}|^2\, F_T\Big(\lambda_{k'} - \frac{k}{T}\Big),
    \end{equation}
    其中 $F_T$ 为 Fejér 核. 这是"QPE 以概率 $|c_{k'}|^2$ 采样到本征相位
    $\lambda_{k'}$ 附近的 $k$"的精确表述: 不同本征分量的贡献按 $|c_{k'}|^2$ 加权, 
    交叉项为零. 这正是基态能量估计与 HHL 中"初态为本征态叠加"情形的分析基础.
\end{Prop}

\subsection{应用: 振幅估计}

振幅放大算子(下一节)的特征值携带成功概率的信息, 因此可以用 QPE 估计概率 ---
这是 QPE 与振幅放大结合的第一个重要应用.

\begin{Prop}[振幅估计]
    设 Grover 算子 $G$ 在二维不变子空间 $\mathrm{span}\{\ket{\mathrm{good}}, \ket{\mathrm{bad}}\}$ 上
    的矩阵表示为旋转角 $\theta$ 的旋转(即 $e^{-i\theta Y}$),
    其两个特征态为 $\ket{\pm} = \frac{1}{\sqrt{2}}(\ket{\mathrm{good}} \pm i\ket{\mathrm{bad}})$,
    对应特征值 $e^{\pm i\theta}$.
    用 QPE 估计 $\theta$ 到加性精度 $\varepsilon$, 即可得到 $p_0 = \sin^2(\theta/2)$ 的估计, 
    总复杂度 $O(1/\varepsilon)$; 而直接\textbf{蒙特卡洛(Monte Carlo)}采样需要 $O(1/\varepsilon^2)$ 次.
    若 $p_0$ 很小需乘性精度, 复杂度为 $O(p_0^{-1/2}\varepsilon^{-1})$, 相对采样的
    $O(p_0^{-1}\varepsilon^{-2})$ 仍是平方级改进.
\end{Prop}

\begin{Proof}[特征态与特征值的验证]
    在基 $\{\ket{\mathrm{good}}, \ket{\mathrm{bad}}\}$ 下 $G$ 的矩阵为
    $[G] = \left(\begin{smallmatrix} \cos\theta & \sin\theta \\ -\sin\theta & \cos\theta \end{smallmatrix}\right)$
    (见定理 \ref{thm:grover} 的推导). 直接作用:
    \[ G\ket{\pm} = \frac{1}{\sqrt{2}}\Big(
       (\cos\theta \mp i\sin\theta)\ket{\mathrm{good}}
       + (\sin\theta \pm i\cos\theta)\ket{\mathrm{bad}}\Big), \]
    利用 $e^{\pm i\theta} = \cos\theta \pm i\sin\theta$ 与
    $\sin\theta \pm i\cos\theta = \pm i e^{\pm i\theta}$, 得
    \[ G\ket{\pm} = e^{\pm i\theta} \frac{1}{\sqrt{2}}\big(\ket{\mathrm{good}} \pm i\ket{\mathrm{bad}}\big)
       = e^{\pm i\theta}\ket{\pm}. \]
    故 $\ket{\pm}$ 是特征态, 特征值为 $e^{\pm i\theta}$.
    又初态 $\ket{\phi_0} = \sqrt{p_0}\ket{\mathrm{good}} + \sqrt{1-p_0}\ket{\mathrm{bad}}$
    满足 $|\braket{\pm}{\phi_0}|^2 = 1/2$(两个分支各半), 因此对 $G$ 做 QPE,
    一次运行即(以概率 $1/2$)测出 $\theta$ 到精度 $\varepsilon$.
\end{Proof}

\subsection{特征值变换与矩阵函数}

QPE 的核心价值不止于"读出相位", 还在于它能作为\textbf{相干的本征值读出原语}
(coherent eigenvalue readout primitive), 用来对量子态施加矩阵函数 $f(H)$.

\begin{Def}[用 QPE 实现矩阵函数]
    设 $H$ 为厄米矩阵, $f: \mathbb{R} \to \mathbb{R}$ 满足 $|f(x)| \le 1$
    ($x \in [-1, 1]$), $U = e^{2\pi i H}$ 由 Hamiltonian 模拟实现. 对
    $H\ket{v_j} = \lambda_j\ket{v_j}$, 输入 $\ket{b} = \sum_j \beta_j\ket{v_j}$,
    目标是制备 $\ket{\psi} \propto f(H)\ket{b} = \sum_j \beta_j f(\lambda_j)\ket{v_j}$.
\end{Def}

\begin{Note}[三步实现: QPE + 受控旋转 + 逆 QPE]
    记整个 QPE 电路(含逆 QFT)为 $U_{\mathrm{QPE}}$. 在"本征值 $\lambda_j$ 恰有 $d$ 位
    二进制表示"的理想假设下:
    \begin{enumerate}
        \item \textbf{QPE 阶段}: $U_{\mathrm{QPE}}\ket{0^d}\ket{v_j} = \ket{\tilde\lambda_j}\ket{v_j}$,
        由线性性, $U_{\mathrm{QPE}}\ket{0^d}\ket{b} = \sum_j \beta_j\ket{\tilde\lambda_j}\ket{v_j}$,
        本征值被\textbf{相干地}编码到相位寄存器;
        \item \textbf{受控旋转阶段}: 对信号比特施加受控旋转
        \[ U_{\mathrm{CR}}\ket{0}\ket{\tilde\lambda_j}
        = \Big(\sqrt{1 - f(\tilde\lambda_j)^2}\,\ket{0} + f(\tilde\lambda_j)\,\ket{1}\Big)\ket{\tilde\lambda_j}, \]
        使信号比特的 $\ket{1}$ 振幅携带 $f(\lambda_j)$;
        \item \textbf{逆 QPE(uncomputation)}: $U_{\mathrm{QPE}}^\dagger$ 把相位寄存器复位为
        $\ket{0^d}$, 丢弃工作寄存器后, 系统与信号比特的联合态为
        \[ U\ket{0}\ket{b} = \sum_j \beta_j\Big(\sqrt{1 - f(\lambda_j)^2}\,\ket{0}
           + f(\lambda_j)\,\ket{1}\Big)\ket{v_j}. \]
        测量信号比特: 结果为 $1$ 时, 系统寄存器的归一化态恰为 $f(H)\ket{b}$.
    \end{enumerate}
    这就是 HHL 算法的核心机制(受控旋转把本征值信息转化为振幅), 也解释了
    HHL 只在信号比特测得 $1$ 时成功的原因. 相比 QSVT, 该路线依赖理想化的
    "本征值精确二进制表示"假设; QSVT 的特征值变换不需要这些假设, 分析更直接
    (见下一模块).
\end{Note}

\section{振幅放大 (AA), 最优振幅放大与非感知振幅放大 (OAA)}
振幅放大是量子算法设计中最重要的工具之一.其核心思想是: 把算法的成功分支与失败分支看成一个二维子空间中的两个方向, 通过两个反射的乘积实现旋转, 逐步把振幅从失败方向转移到成功方向. 相对于经典独立重复采样, 所需操作次数从 $O(1/p)$ 降至 $O(1/\sqrt{p})$, 其中 $p$ 是初始成功概率. 

\subsection{Grover 搜索: 振幅放大的原型}

\begin{Def}[无结构搜索问题]
    给定布尔函数 $f: \{0,1\}^n \to \{0,1\}$ 与 promise: 存在唯一\textbf{标记态(marked state)} $x_0$ 使 $f(x_0) = 1$,
    找出 $x_0$. 经典算法最坏情形需要 $N - 1 = 2^n - 1$ 次查询, 
    量子算法可以做到 $O(\sqrt{N})$ 次 --- 这就是 Grover 的\textbf{二次加速(quadratic speedup)}.
\end{Def}

设 oracle $U_f\ket{x}\ket{y} = \ket{x}\ket{y \oplus f(x)}$. 取 $\ket{y} = \ket{-}$,
由\textbf{相位反冲(phase kickback)}得 $U_f\ket{x}\ket{-} = (-1)^{f(x)}\ket{x}\ket{-}$, 于是\textbf{相位 oracle(phase oracle)}
$U_f\ket{x} = (-1)^{f(x)}\ket{x}$ 等价于\textbf{反射算子(reflection operator)}

\begin{equation}
    R_{x_0} = I - 2\ket{x_0}\bra{x_0}.
\end{equation}

初态取均匀叠加 $\ket{\phi_0} = H^{\otimes n}\ket{0^n}$, 定义关于 $\ket{\phi_0}$ 的反射

\begin{equation}
    R_0 = 2\ket{\phi_0}\bra{\phi_0} - I = H^{\otimes n}\big(2\ket{0^n}\bra{0^n} - I\big)H^{\otimes n}.
\end{equation}

$R_0$ 可以由 $n$ 比特受控 $X$ 门与辅助比特 $\ket{-}$ 实现.

\begin{Eg}[Grover 算法流程]
    Grover 算法(live\_notes 算法 11.1)的完整流程如下 --- 采用具体可执行的形式, 
    比特 oracle 与扩散算子各作用一次构成一次迭代:
    \begin{enumerate}
        \item \textbf{初始化}: 制备 $\ket{0^n}\ket{1}$, 再对\textbf{每个}量子比特施加 Hadamard 门:
        \[ \ket{\psi} \leftarrow H^{\otimes(n+1)}\ket{0^n}\ket{1}
           = \Big(\frac{1}{\sqrt{N}}\sum_{x\in\{0,1\}^n}\ket{x}\Big)\ket{-}
           = \ket{\phi_0}\ket{-}, \]
        其中 $\ket{\phi_0} = H^{\otimes n}\ket{0^n}$ 为均匀叠加态, 
        $\ket{-} = \frac{1}{\sqrt{2}}(\ket{0}-\ket{1})$ 用于相位反冲;
        \item \textbf{标记(oracle)}: 施加比特 oracle $U_f$(定义见上), 由相位反冲
        $U_f\ket{x}\ket{-} = (-1)^{f(x)}\ket{x}\ket{-}$, 对标记态 $\ket{x_0}$ 附加 $-1$ 相位 ---
        等价于反射算子 $R_{x_0} = I - 2\ket{x_0}\bra{x_0}$;
        \item \textbf{扩散(diffusion)}: 施加\textbf{扩散算子}
        \[ D := \big(2\ket{\phi_0}\bra{\phi_0} - I\big)\otimes I
           = H^{\otimes n}\big(2\ket{0^n}\bra{0^n} - I\big)H^{\otimes n}\otimes I, \]
        即关于均匀叠加态的反射 $R_0$(中间的 $2\ket{0^n}\bra{0^n}-I$ 可用辅助比特与受控 $X$ 实现);
        \item \textbf{迭代}: 令 $\theta = 2\arcsin(1/\sqrt{N})$, 重复步骤 2--3 共
        $k = \lfloor \pi/(2\theta) \rfloor$ 次;
        \item \textbf{测量与验证}: 在计算基上测量系统寄存器得到 $x$, 用 $f(x)$ 检验;
        失败则重复.
    \end{enumerate}
    迭代 $k^* = \lfloor \pi/(2\theta) \rfloor = \lfloor \pi/(4\arcsin(1/\sqrt{N})) \rfloor$ 次后, 
    成功概率至少 $1 - 4/N$, 查询数为 $O(\sqrt{N})$; 这两点由下文的
    Grover 迭代定理保证, 结合经典验证即可把失败概率压到任意小的常数.
\end{Eg}

\begin{Thm}[Grover 迭代] \label{thm:grover}
    记 $\theta = 2\arcsin(1/\sqrt{N})$, 分解
    $\ket{\phi_0} = \sin(\theta/2)\ket{x_0} + \cos(\theta/2)\ket{x_0^\perp}$.
    \textbf{Grover 算子(Grover operator)} $G = R_0 R_{x_0}$ 在二维不变子空间
    $\mathrm{span}\{\ket{x_0}, \ket{x_0^\perp}\}$ 内是旋转角为 $\theta$ 的旋转:
    \begin{equation}
        [G] = \begin{pmatrix} \cos\theta & \sin\theta \\ -\sin\theta & \cos\theta \end{pmatrix},
    \end{equation}
    且
    \begin{equation}
        G^k \ket{\phi_0} = \sin\!\big((2k+1)\theta/2\big)\ket{x_0}
        + \cos\!\big((2k+1)\theta/2\big)\ket{x_0^\perp}.
    \end{equation}
\end{Thm}

\begin{Proof}
    在基 $B = \{\ket{x_0}, \ket{x_0^\perp}\}$ 中逐块计算. 由
    $\ket{\phi_0} = \sin(\theta/2)\ket{x_0} + \cos(\theta/2)\ket{x_0^\perp}$,
    其坐标与投影算子在基 $B$ 下为
    \[ \ket{\phi_0} = \begin{pmatrix} \sin(\theta/2) \\ \cos(\theta/2) \end{pmatrix}, \qquad
       \ket{\phi_0}\bra{\phi_0} = \begin{pmatrix} \sin^2(\theta/2) & \sin(\theta/2)\cos(\theta/2) \\
       \sin(\theta/2)\cos(\theta/2) & \cos^2(\theta/2) \end{pmatrix}. \]
    代入 $R_0 = 2\ket{\phi_0}\bra{\phi_0} - I$, 用倍角公式
    \[ 2\sin^2(\theta/2) - 1 = -\cos\theta, \quad
       2\cos^2(\theta/2) - 1 = \cos\theta, \quad
       2\sin(\theta/2)\cos(\theta/2) = \sin\theta, \]
    逐元素化简, 得
    \begin{equation}
        [R_0]_B = \begin{pmatrix} 2\sin^2(\theta/2) - 1 & 2\sin(\theta/2)\cos(\theta/2) \\
        2\sin(\theta/2)\cos(\theta/2) & 2\cos^2(\theta/2) - 1 \end{pmatrix}
        = \begin{pmatrix} -\cos\theta & \sin\theta \\ \sin\theta & \cos\theta \end{pmatrix}.
    \end{equation}
    类似地, 由 $R_{x_0} = I - 2\ket{x_0}\bra{x_0}$ 与 $\ket{x_0} = (1, 0)^T$:
    \begin{equation}
        [R_{x_0}]_B = \begin{pmatrix} -1 & 0 \\ 0 & 1 \end{pmatrix}.
    \end{equation}
    于是
    \begin{equation}
        [G]_B = [R_0R_{x_0}]_B
        = \begin{pmatrix} -\cos\theta & \sin\theta \\ \sin\theta & \cos\theta \end{pmatrix}
        \begin{pmatrix} -1 & 0 \\ 0 & 1 \end{pmatrix}
        = \begin{pmatrix} \cos\theta & \sin\theta \\ -\sin\theta & \cos\theta \end{pmatrix}
        = e^{-i\theta Y},
    \end{equation}
    即 $G$ 是旋转角 $\theta$ 的旋转, 两个反射的乘积确实是旋转(这是"$\mathrm{span}$ 不变"
    的代数证明). 直接作用于初态:
    \begin{equation*}
    \begin{aligned}
        G\ket{\phi_0}
        &= \big(\sin(\theta/2)\cos\theta + \cos(\theta/2)\sin\theta\big)\ket{x_0}
        + \big(\cos(\theta/2)\cos\theta - \sin(\theta/2)\sin\theta\big)\ket{x_0^\perp} \\
        &= \sin(3\theta/2)\ket{x_0} + \cos(3\theta/2)\ket{x_0^\perp},
    \end{aligned}
    \end{equation*}
    即角度从 $\theta/2$ 增为 $3\theta/2$; 旋转 $k$ 次后角度为
    $(2k+1)\theta/2$, 得 $G^k\ket{\phi_0}$ 的表达式. 图 \ref{fig:grover} 给出其几何图像.
\end{Proof}

\begin{Note}[矩阵形式与基的选择]
    旋转矩阵的显式形式依赖于二维不变子空间的\textbf{基}. 两种常用基给出不同形式:
    在本节基 $\{\ket{x_0}, \ket{x_0^\perp}\}$ 下, 
    \[ [G] = \begin{pmatrix} \cos\theta & \sin\theta \\ -\sin\theta & \cos\theta \end{pmatrix}, \]
    而在基 $\{\ket{\phi_0}, \ket{\phi_0'}\}$ 下(见下节振幅放大及 live\_notes 式 (11.14))为
    \[ [G] = \begin{pmatrix} \cos\theta & -\sin\theta \\ \sin\theta & \cos\theta \end{pmatrix}. \]
    两个正交基由行列式为 $-1$ 的\textbf{反射}矩阵联系, 只改变反对角元的符号, 
    表示的旋转动力学($G^k\ket{\phi_0}$ 的成功振幅 $\sin((2k+1)\theta/2)$)完全相同.
\end{Note}

\begin{Cor}[查询复杂度]
    取 $k = O\big(1/\theta\big) = O(\sqrt{N})$ 次迭代, 即可使 $\ket{x_0}$ 的振幅达到
    $\Omega(1)$. 具体地, 测量失败概率不超过 $4/N$, 且输出可以经典验证
    ($f(x)$ 检查), 重复即可. 查询复杂度 $O(\sqrt{N})$ 与经典 $O(N)$ 相比是二次加速;
    且这是最优的 --- 任何量子算法至少需要 $\Omega(\sqrt{N})$ 次查询(见讲义 2.4 节的混合论证证明).
    若存在 $M$ 个标记态, 复杂度为 $O(\sqrt{N/M})$.
\end{Cor}
\begin{Proof}[简略证明]
    \begin{equation*}
        \begin{aligned}
            G^k H^{\otimes n}\ket{0} &= \sin((2k+1)\theta/2)\ket{x_0} + \cos((2k+1)\theta/2)\ket{x_0^\perp}, \\
            \theta &= 2\arcsin(1/\sqrt{N}) \sim 1/\sqrt{N}, \\
        \end{aligned}
    \end{equation*}
    为了$\sin((2k+1)\theta/2)\approx 1$, 取
    \[(2k+1)\theta/2\approx \pi/2\]
    因此次数
    \[k\sim 1/\theta\sim \sqrt{N}\]
\end{Proof}

\begin{Note}[与经典搜索的对比]
    无序搜索中经典\textbf{确定性}算法最坏情形需要 $N-1$ 次查询;
    \textbf{随机化}经典算法(随机猜测 + 验证)平均也需要 $\Theta(N)$ 次
    (单次成功概率 $1/N$, 期望 $N$ 次尝试). Grover 的 $O(\sqrt{N})$ 是
    量子在查询复杂度上的严格二次加速: 下界 $\Omega(\sqrt{N})$ 的混合论证
    说明, 经典每次查询"线性"地更新概率, 而量子查询以旋转的方式更新振幅, 
    信息以 $\sqrt{p}$ 而非 $p$ 的速度增长. 有 $M$ 个标记态时, 
    经典仍需 $\Theta(N/M)$ 次查询, 量子为 $O(\sqrt{N/M})$
    (当 $M \ll N$ 时仍为二次加速; 当 $M$ 接近 $N/2$ 时优势消失).
\end{Note}

\begin{Proof}[经典下界: 至少 $N-1$ 次查询]
    反设存在无偏经典算法最坏情形用至多 $N-2$ 次查询即可识别标记元素 $x_0$.
    把查询补到恰好 $N-2$ 次(已知只有一个标记, 后续查询不提供新信息).
    则存在两个\textbf{未被查询}的候选值 $x, x' \notin \{x_1, \dots, x_{N-2}\}$.
    若真实标记是 $x$, 则所有已查询位置的输出均为 $0$; 若真实标记是 $x'$,
    观察到的证据完全相同(同样全部为 $0$). 两种假设下的数据无差别, 因此任何
    以概率 $> 1/2$ 正确识别的估计子必然偏向二者之一, 矛盾. 故最坏情形至少需要
    $N-1$ 次查询(live\_notes 命题 11.1). 量子下界 $\Omega(\sqrt{N})$ 的混合论证
    参见讲义第 2.4 节.
\end{Proof}

\subsection{振幅放大}

Grover 搜索只是振幅放大在"标记态 = 单个计算基态"时的特例.

\begin{Thm}[振幅放大] \label{thm:AA}
    设 $\Pi_{\mathrm{good}}$ 是"好空间"上的投影, 初态由 oracle $U_0$ 制备:
    \begin{equation}
        \ket{\phi_0} = U_0 \ket{0^n}, \qquad
        \ket{\phi_0} = \Pi_{\mathrm{good}}\ket{\phi_0} + (I - \Pi_{\mathrm{good}})\ket{\phi_0}
        = \sqrt{p}\,\ket{\mathrm{good}} + \sqrt{1-p}\,\ket{\mathrm{bad}},
    \end{equation}
    其中 $p = \bra{\phi_0}\Pi_{\mathrm{good}}\ket{\phi_0}$, $\ket{\mathrm{good}}$,
    $\ket{\mathrm{bad}}$ 正交, 记 $p = \sin^2(\theta/2)$.
    假设可以访问反射 $R_{\mathrm{good}} = I - 2\Pi_{\mathrm{good}}$ 与
    $R_0 = 2\ket{\phi_0}\bra{\phi_0} - I = U_0(2\ket{0^n}\bra{0^n} - I)U_0^\dagger$,
    定义 $G = R_0 R_{\mathrm{good}}$, 则对任意 $k \ge 0$,
    \begin{equation}
        \big|\bra{\mathrm{good}} G^k \ket{\phi_0}\big| = \sin\!\big((2k+1)\theta/2\big).
    \end{equation}
    于是 $k = O(1/\theta) = O(1/\sqrt{p})$ 次迭代后, 成功概率达到 $\Omega(1)$.
\end{Thm}

\begin{Proof}
    记与 $\ket{\phi_0}$ 正交的单位向量
    \begin{equation}
        \ket{\phi_0^\perp} = \cos(\theta/2)\ket{\mathrm{good}} - \sin(\theta/2)\ket{\mathrm{bad}},
    \end{equation}
    则 $\{\ket{\phi_0}, \ket{\phi_0^\perp}\}$ 张成同一个二维子空间
    (由 $\ket{\phi_0}$, $\ket{\mathrm{good}}$ 生成). 反射作用为
    \begin{equation}
        R_{\mathrm{good}}\ket{\mathrm{good}} = -\ket{\mathrm{good}}, \quad
        R_{\mathrm{good}}\ket{\mathrm{bad}} = \ket{\mathrm{bad}}, \quad
        R_0\ket{\phi_0} = \ket{\phi_0}, \quad
        R_0\ket{\phi_0^\perp} = -\ket{\phi_0^\perp}.
    \end{equation}
    用 $\ket{\phi_0}$, $\ket{\phi_0^\perp}$ 展开 $\ket{\mathrm{good}} = \sin(\theta/2)\ket{\phi_0}
    + \cos(\theta/2)\ket{\phi_0^\perp}$, $\ket{\mathrm{bad}} = \cos(\theta/2)\ket{\phi_0}
    - \sin(\theta/2)\ket{\phi_0^\perp}$, 直接计算
    \begin{equation}
        [G] = \begin{pmatrix} \cos\theta & -\sin\theta \\ \sin\theta & \cos\theta \end{pmatrix}
        \quad \text{于基 } \{\ket{\phi_0}, \ket{\phi_0^\perp}\},
    \end{equation}
    故 $G^k\ket{\phi_0} = \cos(k\theta)\ket{\phi_0} + \sin(k\theta)\ket{\phi_0^\perp}$, 于是
    \begin{equation}
        \bra{\mathrm{good}}G^k\ket{\phi_0}
        = \sin(\theta/2)\cos(k\theta) + \cos(\theta/2)\sin(k\theta)
        = \sin\!\big((2k+1)\theta/2\big),
    \end{equation}
    其中最后一步用了正弦和角公式. Grover 搜索是其特例:
    $\ket{\mathrm{good}} = \ket{x_0}$, $p = 1/N$, $\theta = 2\arcsin(1/\sqrt{N})$.
\end{Proof}

\begin{Eg}[信号比特场景]
    若 $U_0$ 需要 $m$ 个辅助(信号)比特, 即
    \[ U_0 \ket{0^m}\ket{0^n} = \sqrt{p}\,\ket{0^m}\ket{\phi} + \sqrt{1-p}\,\ket{\phi^\perp}, \]
    其中 $\ket{\phi^\perp}$ 与 $\ket{0^m}$ 正交. 此时"好"的状态由辅助比特全 $0$ 标志, 
    反射简化为 $R_{\mathrm{good}} = (I - 2\ket{0^m}\bra{0^m}) \otimes I$, \textbf{后选择(post-selection)}测量辅助比特
    即得到成功概率 $p$. 这类"可验证成功"的结构在 block-encoding 等场景中反复出现.
\end{Eg}

\begin{Note}[为什么是二次加速: $O(1/\sqrt{p})$ 的由来]
    初始成功概率 $p = \sin^2(\theta/2)$, 即初始成功\textbf{振幅}为
    $\sin(\theta/2) \approx \sqrt{p}$. 由振幅放大定理, 迭代 $k$ 次后的成功振幅为
    $a_k = \sin\big((2k+1)\theta/2\big)$. 推导 $O(1/\sqrt{p})$ 需要三步:
    \begin{enumerate}
        \item \textbf{旋转角与初始振幅同阶}: $\theta = 2\arcsin\sqrt{p} \approx 2\sqrt{p}$,
        每次迭代在二维子空间内把状态旋转固定的角度 $\theta$;
        \item \textbf{振幅线性增长}: 在角度 $(2k+1)\theta/2$ 尚未越界时, 
        \[ a_k = \sin\!\Big(\frac{(2k+1)\theta}{2}\Big) \approx \frac{(2k+1)\theta}{2} \approx k\theta, \]
        即每步振幅近似增加 $\theta \approx 2\sqrt{p}$;
        \item \textbf{概率二次增长}: 成功概率是振幅的平方, 
        \[ P_k = a_k^2 \approx k^2\theta^2 = O\big(k^2 p\big), \]
        故把成功概率提升到常数只需
        \[ k = O\Big(\frac{1}{\theta}\Big) = O\Big(\frac{1}{\arcsin\sqrt{p}}\Big) = O\Big(\frac{1}{\sqrt{p}}\Big) \]
        次迭代(严格地, 最大值在 $(2k+1)\theta/2 = \pi/2$, 即 $k \approx \pi/(2\theta)$).
    \end{enumerate}
    对比经典: 成功概率为 $p$ 的伯努利试验, 成功概率按\textbf{线性}速率增长
    ($P \sim kp$), 期望重复 $O(1/p)$ 次. 差距的根源在于量子以\textbf{振幅}
    $\sqrt{p}$(而非概率 $p$)为基本量: 概率是振幅的平方, 振幅每步增加
    $O(\sqrt{p})$ 就使概率每步增加 $O(p)$ --- 用振幅的线性增长换取概率的
    二次增长, 这正是"量子处理振幅而非概率"带来的平方级改进.
\end{Note}

\begin{Note}[$W$ 的本征结构]
    由 $[W]_{\phi_0,\phi_0'} = e^{-i\theta Y}$(旋转角 $\theta = 2\arcsin\sqrt{p}$),
    $W$ 在二维不变子空间 $\mathrm{span}\{\ket{\phi_0}, \ket{\phi_0'}\}$ 内的特征向量为
    \[ \ket{\pm} = \frac{1}{\sqrt{2}}\big(\ket{\phi_0} \pm i\ket{\phi_0'}\big),
       \qquad W\ket{\pm} = e^{\mp i\theta}\ket{\pm}. \]
    这正是振幅估计(见 QPE 一章)的出发点: 成功概率 $p$ 被编码进 $W$ 的
    本征相位 $e^{\pm i\theta}$, 用 QPE 即可对 $p$ 做平方级加速估计.
\end{Note}

\begin{Note}[$p$ 未知时的策略]
    若 $p$ 未知, 无法精确选取迭代次数. 经典策略(live\_notes §11.2)是均匀随机选取
    $m \in [1, 2^r - 1]$, 执行 $G^m$ 后用 $R_{\mathrm{good}}$ 验证; 当 $2^r \gtrsim \pi/\theta$
    时, 迭代角度在单位圆上"随机采样", $\cos^2$ 的平均值为 $1/2$, 故成功概率为常数, 
    重复次数按几何分布, 期望 $O(1/\sqrt{p})$. 现代方法则用基于 QSVT 的
    \textbf{固定点振幅放大}(fixed-point amplitude amplification), 在 $p$ 未知时
    也能把成功概率逼近 $1$(见下一模块).
\end{Note}

\subsection{非感知振幅放大 (Oblivious amplitude amplification, OAA)}

\textbf{非感知振幅放大}(oblivious amplitude amplification, OAA)是振幅放大的一种推广形式.
它的主要特点是: 振幅放大的反射操作不依赖于数据寄存器中的未知输入态, 
而只依赖于辅助量子比特(ancilla)的状态, 因此可以对任意未知输入态统一地进行成功振幅放大.

\subsubsection{基本设定}

设数据寄存器中的输入态为 $\ket{\psi}$, 辅助寄存器初始化为 $\ket{0^a}$.
考虑一个酉算子 $W$, 满足
\[
    W\ket{0^a}\ket{\psi}
    = \sin\theta\,\ket{0^a}\ket{\psi_{\mathrm{good}}}
    + \cos\theta\,\ket{\Phi^\perp_\psi},
    \qquad \braket{0^a}{\Phi^\perp_\psi} = 0.
\]
第一项对应成功子空间, 第二项对应失败子空间, 成功概率为 $p = \sin^2\theta$.
因此也可以写成
\[
    W\ket{0^a}\ket{\psi}
    = \sqrt{p}\,\ket{0^a}\ket{\psi_{\mathrm{good}}}
    + \sqrt{1-p}\,\ket{\Phi^\perp_\psi}.
\]
OAA 的目标是通过若干次酉操作, 将成功分量 $\ket{0^a}\ket{\psi_{\mathrm{good}}}$ 的振幅增大, 
而不需要知道输入态 $\ket{\psi}$ 的具体形式.

\subsubsection{成功子空间的投影与反射}

定义辅助寄存器处于 $\ket{0^a}$ 时对应的投影算子
\[
    P = \ket{0^a}\bra{0^a} \otimes I,
\]
其中 $I$ 是作用在数据寄存器上的单位算子. 对任意数据态 $\ket{\varphi}$,
\[
    P\ket{0^a}\ket{\varphi} = \ket{0^a}\ket{\varphi},
\]
而对与 $\ket{0^a}$ 正交的辅助寄存器状态, $P\ket{\Phi^\perp} = 0$.
由投影算子 $P$ 定义反射算子
\[
    R = 2P - I,
\]
于是 $R\ket{0^a}\ket{\varphi} = \ket{0^a}\ket{\varphi}$, 而对失败子空间中的状态
$R\ket{\Phi^\perp} = -\ket{\Phi^\perp}$.
由于 $P = \ket{0^a}\bra{0^a}\otimes I$ 在数据寄存器上仅作用单位算子 $I$,
该反射与具体的数据态 $\ket{\psi}$ 无关 --- 这正是 ``oblivious'' 的含义.

\subsubsection{二维不变子空间}

定义 $\ket{\Psi} = \ket{0^a}\ket{\psi}$. 经过 $W$ 后, 
\[
    W\ket{\Psi} = \sin\theta\,\ket{G} + \cos\theta\,\ket{B},
\]
其中 $\ket{G} = \ket{0^a}\ket{\psi_{\mathrm{good}}}$ 表示成功分量, 
$\ket{B} = \ket{\Phi^\perp_\psi}$ 表示失败分量, 二者满足 $\braket{G}{B} = 0$.
因此, 尽管完整 Hilbert 空间可能具有很高的维数, OAA 的动力学实际上被限制在二维子空间
\[
    \mathcal{K} = \mathrm{span}\{\ket{G}, \ket{B}\}
\]
中. 这使得 OAA 可以被理解为该二维平面中的旋转过程.

\subsubsection{OAA 迭代算子}

除了关于成功子空间的反射 $R$ 外, 通过酉算子 $W$ 还可以构造另一个反射 $WRW^\dagger$.
OAA 的基本迭代算子定义为
\[
    Q = -WRW^\dagger R,
\]
这里 $Q$ 是两个反射算子的乘积. 需要注意的是, 不同文献可能采用不同的反射符号约定, 
例如也常定义 $R' = I - 2P$. 这种差异只会改变整体相位或旋转方向, 
不会改变振幅放大的基本结论.

\subsubsection{两个反射产生二维旋转}

初始状态为
\[
    W\ket{\Psi} = \sin\theta\,\ket{G} + \cos\theta\,\ket{B}.
\]
在二维子空间 $\mathcal{K} = \mathrm{span}\{\ket{G}, \ket{B}\}$ 中, 
OAA 迭代算子 $Q$ 等价于一次旋转: 每施加一次 $Q$, 量子态在该二维平面中旋转角度 $2\theta$.
因此, 一次 OAA 迭代后
\[
    QW\ket{\Psi} = \sin(3\theta)\,\ket{G} + \cos(3\theta)\,\ket{B},
\]
两次迭代后
\[
    Q^2W\ket{\Psi} = \sin(5\theta)\,\ket{G} + \cos(5\theta)\,\ket{B}.
\]
一般地, 经过 $k$ 次 OAA 迭代后
\[
    \boxed{Q^kW\ket{\Psi} = \sin\big((2k+1)\theta\big)\ket{G} + \cos\big((2k+1)\theta\big)\ket{B}},
\]
因此成功概率变为
\[
    \boxed{p_k = \sin^2\big((2k+1)\theta\big)}.
\]

\subsubsection{振幅放大的复杂度}

初始成功概率为 $p = \sin^2\theta$. 当 $p \ll 1$ 时, $\sin\theta \approx \theta$,
因此 $\theta \approx \sqrt{p}$. 为了使成功概率接近 $1$, 需要令
$(2k+1)\theta \approx \pi/2$, 由此得到
\[
    k \approx \frac{\pi}{4\theta} - \frac12.
\]
由于 $\theta \approx \sqrt{p}$, 因此
\[
    \boxed{k = O\Big(\frac{1}{\sqrt{p}}\Big)}.
\]
如果直接重复运行并测量, 为了获得一次成功结果通常需要 $O(1/p)$ 次尝试.
因此 OAA 实现了从 $O(1/p)$ 到 $O(1/\sqrt{p})$ 的平方加速.

\subsubsection{OAA 与普通振幅放大的区别}

普通振幅放大中, 反射操作可能依赖于具体的输入态, 例如可能需要构造关于
$\ket{0^a}\ket{\psi}$ 的反射. 若 $\ket{\psi}$ 是未知的, 这种反射通常难以直接实现.
OAA 则只利用辅助寄存器定义成功子空间 $P = \ket{0^a}\bra{0^a}\otimes I$;
由于数据寄存器上作用的是单位算子 $I$, 这一投影以及相应的反射算子都不依赖于
$\ket{\psi}$ 的具体形式. 换句话说, OAA 可以对任意未知输入态 $\ket{\psi}$
使用同一套振幅放大电路. 因此, OAA 的核心特征可以概括为
\[
    \boxed{\text{Amplify the success amplitude without knowing the input data state.}}
\]

\subsubsection{OAA 的数学本质}

OAA 的数学本质是两个反射算子的乘积. 对于投影算子 $P$, 定义反射 $R_P = 2P - I$;
经过酉变换 $W$ 后, 另一个反射为 $WR_PW^\dagger$. 于是 OAA 迭代算子为
\[
    Q = -\big(WR_PW^\dagger\big)R_P.
\]
两个反射的乘积在相应的二维不变子空间中等价于一次旋转, 因此可以概括为
\[
    \boxed{\text{reflection} + \text{reflection} = \text{rotation}}.
\]
因此, OAA 并不是直接修改测量概率, 而是通过量子干涉不断旋转量子态, 
使其逐渐靠近成功子空间, 从而增大成功分量的测量概率.

\subsubsection{与 block-encoding 的联系}

上面的叙述用的是"状态编码"的表述; 在 block-encoding 的框架下, 同一走算子有更重要的应用.
block-encoding 的成功概率受归一化常数限制: 若 $V$ 是矩阵 $A$ 的
$(\alpha, m, 0)$-block-encoding, 则其作用在态 $\ket{\psi}$ 上"成功"(落到 $\ket{0^m}$ 分支)的概率
仅为 $\norm{A\ket{\psi}}^2/\alpha^2$. 用普通振幅放大需要关于\textbf{初态}
$V\ket{0^m}\ket{\psi}$ 的反射 $R_0$, 多次使用时要求"回退"到初态, 代价巨大; 且
成功/失败本身携带了关于 $\ket{\psi}$ 的信息(似然比
$\norm{A\ket{\psi}}^2/\norm{A\ket{\psi'}}^2$ 对一般态可能远离 $1$), 阻碍链式使用.
OAA 通过只关于\textbf{好空间}与"包含初态的子空间"反射来绕开这两个问题.

\begin{Def}[OAA 走算子]
    设 $\Pi_{\mathrm{good}}$ 是辅助寄存器上的投影算子(标志"好"子空间),
    $V$ 是酉 $U$ 的 $(\alpha, m, 0)$-block-encoding, 即
    \[ (\Pi_{\mathrm{good}}\otimes I)\, V\, (\Pi_{\mathrm{good}}\otimes I) = \frac{U}{\alpha}, \]
    (标准 block-encoding 取 $\Pi_{\mathrm{good}} = \ket{0^m}\bra{0^m}$, 即上面的
    $P = \ket{0^a}\bra{0^a}\otimes I$). 记 $Z_{\mathrm{good}} := I - 2\Pi_{\mathrm{good}}$,
    定义\textbf{OAA 走算子}
    \begin{equation}
        W := -V\,(Z_{\mathrm{good}}\otimes I)\, V^\dagger\,(Z_{\mathrm{good}}\otimes I),
    \end{equation}
    这正是上面迭代算子 $Q = -WRW^\dagger R$ 的同一构造($V$ 对应上面的 $W$,
    $Z_{\mathrm{good}}\otimes I = -(2P - I) = -R$, 两个负号相消).
\end{Def}

\begin{Thm}[非感知振幅放大]
    在上述记号下, 对任意满足 $\Pi_{\mathrm{good}}\ket{\mathrm{good}} = \ket{\mathrm{good}}$
    的态,
    \begin{equation}
        (\Pi_{\mathrm{good}}\otimes I)\, W^k\, V\,(\ket{\mathrm{good}}\otimes I)
        = U\,\sin\!\big((2k+1)\arcsin(1/\alpha)\big),
    \end{equation}
    代价为 $2k+1$ 次 $V$ 与 $2k$ 次 $Z_{\mathrm{good}}$. 定义 $\sin\theta = 1/\alpha$
    ($\alpha \ge 1$), 则 $W$ 在"好空间"内是旋转角 $2\theta$ 的旋转, 迭代
    $k = O(\alpha)$ 次即可把成功概率提升到常数.
\end{Thm}

\begin{Note}
    关键性质: 反射 $V(Z_{\mathrm{good}}\otimes I)V^\dagger$ 只依赖好空间与 $V$,
    \textbf{不依赖具体输入态} --- 这正是"非感知"(oblivious)的含义. 它是
    Hamiltonian 模拟中 LCU/block-encoding 方案把指数级衰减的成功概率恢复到
    常数的基础, 也是下一模块 \textbf{qubitization} 的前奏:
    $W$ 正是 qubitization 在 block-encoding 上对应的迭代算子.
\end{Note}

\subsection{振幅放大的应用}

\begin{Note}[多阶段中间测量的算法]
    考虑一个含有多步中间测量的算法, 只有全部辅助比特都测得 $0$ 才算成功
    (例如 block-encoding 的多步组合). 设第 $i$ 步的成功概率为 $p_i$
    (条件于之前各步成功), 则累计成功概率 $p = p_1 \cdots p_L$,
    直接重复需要 $O(1/p)$ 次. 由测量延迟原理, 用 $L-1$ 个辅助比特把
    中间测量推迟, 化为相干过程, 即可构造关于"信号比特"的反射算子, 
    应用振幅放大把重复次数降到 $O(1/\sqrt{p})$ --- 对 $p$ 的平方级改进.
    更进一步, \textbf{压缩装置}(compression gadget, live\_notes 例 11.10)
    用一个 $m = \lceil \log_2(L+1) \rceil$ 比特的计数器寄存器(模 $M = 2^m$ 的
    加法电路 $ADD$)记录成功次数, 把辅助比特数从 $L-1$ 降到 $O(\log L)$.
\end{Note}

\begin{Note}[与 NP 类问题的联系]
    振幅放大可自然地应用于\textbf{任意 NP 问题}: 对"是"实例, 解的验证 witness
    可作为"好空间"的标志, 于是暴力搜索的 $O(1/p)$ 次尝试被降为 $O(1/\sqrt{p})$ 次 ---
    这是"量子相对暴力经典方法的二次加速"的一般形式, 也解释了量子算法对组合优化
    (如 3-SAT)的兴趣来源.
\end{Note}

\section{Lie--Trotter 与哈密顿量模拟}
\subsection{问题设定}

\begin{Def}[哈密顿量模拟]
    给定厄米矩阵 $H$(哈密顿量)与初态 $\ket{\psi(0)}$, 目标是制备
    \begin{equation}
        \ket{\psi(t)} = e^{-iHt}\ket{\psi(0)}.
    \end{equation}
    这是 Schrödinger 方程 $i\partial_t \ket{\psi(t)} = H\ket{\psi(t)}$ 的解.
\end{Def}

\begin{Note}
    哈密顿量模拟是 Feynman 1982 年提出量子计算机设想的直接动机, 
    也是 QPE 及其应用(基态能量估计, HHL 等)中 $U = e^{-iHt}$ 的实现途径.
    与一般线性 ODE 不同, 由于 $e^{-iHt}$ 是酉算子, 我们无需存储态的完整历史, 
    只需关注 $t$ 时刻的状态. 本模块介绍最经典的方法 --- \textbf{乘积公式(product formula)},
    即 Lie--Trotter 分解; 更先进的方法(block-encoding, qubitization, QSVT)见下一模块.
\end{Note}

\subsection{Lie 乘积公式与一阶 Trotter}

当 $H = H_1 + H_2$ 且 $[H_1, H_2] \neq 0$ 时, $e^{-itH} \neq e^{-itH_1}e^{-itH_2}$,
但\textbf{Lie 乘积公式(Lie product formula)}给出

\begin{equation}
    e^{-itH} = \lim_{L \to \infty} \Big(e^{-i(t/L)H_1} e^{-i(t/L)H_2}\Big)^{L}.
\end{equation}

记 $\tau = t/L$ 为步长, 一步的\textbf{局部误差(local error)}为 $O(\tau^2)$:

\begin{Thm}[一阶 Trotter 公式]
    \begin{equation}
        \norm{e^{-i\tau H} - e^{-i\tau H_1} e^{-i\tau H_2}} = O(\tau^2),
    \end{equation}
    其中常数与 $\norm{H_1}$, $\norm{H_2}$ 有关. 对总时间 $t$ 做 $L$ 步, \textbf{全局误差(global error)}
    \begin{equation}
        \norm{e^{-itH} - \big(e^{-i\tau H_1} e^{-i\tau H_2}\big)^{L}} = O(L\tau^2)
        = O\!\Big(\frac{t^2}{L}\Big),
    \end{equation}
    因此达到精度 $\varepsilon$ 需要
    \begin{equation}
        L = O\!\Big(\frac{t^2}{\varepsilon}\Big)
    \end{equation}
    步, 每步调用 $e^{-i\tau H_1}$, $e^{-i\tau H_2}$ 各一次.
\end{Thm}

\begin{Proof}[局部误差的推导(Taylor 积分余项)]
    记 $\widetilde U(\tau) = e^{-i\tau H_1}e^{-i\tau H_2}$. 用带积分余项的 Taylor 展开, 
    逐矩阵元素求导: 精确演化 $U(\tau) = e^{-i\tau H}$ 满足 $U(0) = I$, $U'(0) = -iH$,
    \[ U(\tau) = I - i\tau H - \int_0^{\tau} e^{-isH}H^2(\tau-s)\,ds; \]
    近似演化满足 $\widetilde U(0) = I$, $\widetilde U'(0) = -iH$,
    $\widetilde U''(0) = -e^{-isH_1}(H_1^2 + 2H_1H_2 + H_2^2)e^{-isH_2}\big|_{s=0}
    = -(H_1^2 + 2H_1H_2 + H_2^2)$, 故
    \[ \widetilde U(\tau) = I - i\tau H
       - \int_0^{\tau} e^{-isH_1}(H_1^2 + 2H_1H_2 + H_2^2)e^{-isH_2}(\tau-s)\,ds. \]
    相减并用三角不等式(各 $e^{-i\cdot}$ 因子范数为 1):
    \begin{equation}
        \norm{U(\tau) - \widetilde U(\tau)} \le
        \big(\norm{H}^2 + \norm{H_1}^2 + 2\norm{H_1}\norm{H_2} + \norm{H_2}^2\big)
        \int_0^{\tau}(\tau-s)\,ds
        \le \tau^2\big(\norm{H_1} + \norm{H_2}\big)^2,
    \end{equation}
    其中最后一步用了 $\norm{H} \le \norm{H_1} + \norm{H_2}$ 与 $\int_0^\tau(\tau-s)ds = \tau^2/2$.
    这给出局部误差 $O(\tau^2)$ 及显式常数 $\big(\norm{H_1}+\norm{H_2}\big)^2$.
    $L$ 步复合的全局误差 $O(L\tau^2) = O(t^2/L)$ 来自 Module1 的混合论证
    (每步误差线性累积). 图 \ref{fig:trotter} 给出乘积公式的结构示意.
\end{Proof}

\subsection{交换子型误差界}

上面的 $O(\tau^2)$ 界中的常数可以精确刻画, 而且与 $\norm{H_1}+\norm{H_2}$ 相比, 
更好的常数是\textbf{交换子范数(commutator norm)} $\norm{[H_1, H_2]}$.

\begin{Thm}[交换子型误差界]
    对一阶 Trotter,
    \begin{equation}
        \norm{e^{-i\tau H} - e^{-i\tau H_1} e^{-i\tau H_2}}
        \le \frac{\tau^2}{2}\,\norm{[H_1, H_2]} \le (\tau \alpha)^2,
    \end{equation}
    其中 $\alpha = \max\{\norm{H_1}, \norm{H_2}\}$, 第二式来自
    \begin{equation}
        \norm{[H_1,H_2]} \le 2\norm{H_1}\,\norm{H_2} \le 2\alpha^2.
    \end{equation}
    于是达到精度 $\varepsilon$ 所需的全局步数为
    $L = O\big(t^2\norm{[H_1,H_2]}/\varepsilon\big)$.
\end{Thm}

\begin{Proof}
    记 $\widetilde U(\tau) = e^{-i\tau H_1}e^{-i\tau H_2}$. 直接求导并拆项:
    \begin{equation}
        i\partial_\tau \widetilde U(\tau)
        = H_1 e^{-i\tau H_1}e^{-i\tau H_2} + e^{-i\tau H_1}H_2 e^{-i\tau H_2}
        = H\,\widetilde U(\tau) + \big[e^{-i\tau H_1},\, H_2\big]e^{-i\tau H_2},
    \end{equation}
    其中 $H = H_1 + H_2$, 最后一步用了 $H_2e^{-i\tau H_1} - e^{-i\tau H_1}H_2$
    $= [H_2, e^{-i\tau H_1}]$. 这是含"源项" $[e^{-i\tau H_1},H_2]e^{-i\tau H_2}$ 的方程;
    由 Module1 的 \textbf{Duhamel 原理(Duhamel's principle)}($\widetilde U(0) = U(0) = I$),
    \begin{equation}
        \widetilde U(\tau) - e^{-i\tau H}
        = -i\int_0^{\tau} e^{-iH(\tau-s)}\big[e^{-isH_1},\, H_2\big]e^{-isH_2}\,ds,
    \end{equation}
    因此
    \begin{equation}
        \norm{\widetilde U(\tau) - e^{-i\tau H}}
        \le \int_0^{\tau}\norm{[e^{-isH_1}, H_2]}\,ds.
    \end{equation}
    最后估计交换子. 令
    \begin{equation}
        G(s) := [e^{-isH_1}, H_2]\,e^{isH_1}
        = e^{-isH_1}H_2 e^{isH_1} - H_2,
    \end{equation}
    则 $G(0) = 0$, 且对其求导并积分:
    \begin{equation}
        i\partial_s G(s) = e^{-isH_1}[H_1, H_2]e^{isH_1}, \qquad
        \norm{G(s)} \le s\,\norm{[H_1, H_2]},
    \end{equation}
    又 $\norm{[e^{-isH_1}, H_2]} = \norm{G(s)} \le s\norm{[H_1,H_2]}$, 代入上面的积分:
    \[ \norm{\widetilde U(\tau) - e^{-i\tau H}}
       \le \norm{[H_1,H_2]}\int_0^{\tau} s\,ds = \frac{\tau^2}{2}\norm{[H_1,H_2]}. \]
    这正是定理的常数. 全局误差 $L$ 步线性累积后即得 $L$ 的依赖.
\end{Proof}

\begin{Eg}[横向场 Ising 模型: 交换子界优于算子范数界]
    一维\textbf{横向场 Ising 模型(transverse-field Ising model)}
    \[ H = -J\sum_{i=1}^{n-1} Z_i Z_{i+1} - g\sum_{i=1}^{n} X_i =: H_1 + H_2. \]
    $H_1$ 内部各项两两对易, $H_2$ 内部亦然, 故 $e^{-i\tau H_1}$, $e^{-i\tau H_2}$ 可精确实现
    (乘积分解为局部旋转). 但 $\norm{H_1} = O(n)$, $\norm{H_2} = O(n)$, \textbf{算子范数(operator norm)}界给出
    $\alpha^2 = O(n^2)$, 即 $L = O(n^2 t^2/\varepsilon)$; 而
    $[Z_i Z_{i+1}, X_k] \neq 0$ 仅当 $k = i$ 或 $k = i+1$, 故
    $\norm{[H_1, H_2]} = O(n)$, 交换子界给出更紧的 $L = O(n t^2/\varepsilon)$.
    这说明: 误差界应尽量用交换子范数, 而不是简单的算子范数.
\end{Eg}

\begin{Note}[向量范数界]
    算子范数界对最坏情形的初态成立; 若关心特定初态 $\ket{\psi_0}$, 误差可按
    \begin{equation}
        \frac{t^2}{2}\,\max_{0 \le s \le t}\norm{[H_1,H_2]\ket{\psi(s)}}
    \end{equation}
    估计, 往往比算子范数界小得多(例如势场问题中可与系统规模无关).
\end{Note}

\subsection{对称分裂与高阶 Suzuki 公式}

\begin{Thm}[二阶 Trotter(对称分裂, Strang 分裂)]
    \begin{equation}
        \norm{e^{-i\tau H} - e^{-i\tau H_1/2} e^{-i\tau H_2} e^{-i\tau H_1/2}} = O(\tau^3),
    \end{equation}
    达到精度 $\varepsilon$ 需要 $L = O(t^{3/2} \varepsilon^{-1/2})$ 步, 每步 3 次局部演化.
\end{Thm}

\begin{Thm}[高阶 Suzuki 公式]
    高阶公式可以通过\textbf{组合方法(composition)}递归构造. 若 $U_p$ 是 $p$ 阶公式
    (局部误差 $O(\tau^{p+1})$), 且系数满足 $\sum_j s_j = 1$, $\sum_j s_j^{p+1} = 0$,
    则组合 $U_p(s_m\tau)\cdots U_p(s_1\tau)$ 至少是 $p+1$ 阶.
    特别地, \textbf{Trotter--Suzuki 递推(Trotter--Suzuki recursion)}给出 $2k$ 阶公式
    \begin{equation}
        U_{2k}(\tau) = U_{2k-2}(s_k\tau)\; U_{2k-2}\big((1-4s_k)\tau\big)\; U_{2k-2}(s_k\tau),
        \qquad s_k = \frac{1}{4 - 4^{1/(2k-1)}},
    \end{equation}
    每个 $U_{2k}$ 调用 5 次 $U_{2k-2}$, 故 $U_{2k}$ 含至多 $5^{k-1}$ 次 $U_2$.
    一般地, $p$ 阶公式达到精度 $\varepsilon$ 需要
    \begin{equation}
        L = O\!\big(t^{1+1/p}\,\varepsilon^{-1/p}\big)
    \end{equation}
    步; 当 $p \to \infty$ 时渐近为 $L = O(t^{1+o(1)}\varepsilon^{-o(1)})$.
\end{Thm}

\begin{Lem}[组合方法提升阶数]
    设 $U_p$ 是 $p$ 阶公式, 即短时误差形如 $U_p(t) = U(t) + C\,t^{p+1} + O(t^{p+2})$
    ($U(t) = e^{-itH}$, $C$ 为有界算子). 若实系数 $s_1, \dots, s_m$ 满足
    \begin{equation}
        \sum_{j=1}^{m} s_j = 1, \qquad \sum_{j=1}^{m} s_j^{p+1} = 0,
    \end{equation}
    则组合公式 $U_p^{\mathrm{comp}}(t) := U_p(s_m t)\cdots U_p(s_1 t)$ 至少是 $p+1$ 阶.
\end{Lem}

\begin{Proof}
    记 $U_j := U(s_j t)$, $\widetilde U_j := U_p(s_j t) = U_j + C(s_j t)^{p+1} + O(t^{p+2})$.
    由群性质 $\prod_j U_j = e^{-itH\sum_j s_j} = e^{-itH} = U(t)$(用了 $\sum s_j = 1$),
    且每个 $\widetilde U_j = U_j\big(I + O(t)\big)$, 相乘时保留到 $t^{p+1}$ 阶:
    \begin{equation}
        U_p^{\mathrm{comp}}(t) = \prod_{j=1}^{m} \Big(U_j + C(s_j t)^{p+1}\Big)
        = U(t) + C\,t^{p+1}\Big(\sum_{j=1}^{m} s_j^{p+1}\Big) + O(t^{p+2}),
    \end{equation}
    由 $\sum_j s_j^{p+1} = 0$ 消去 $t^{p+1}$ 项, 故误差为 $O(t^{p+2})$, 阶数至少 $p+1$.
    Trotter--Suzuki 递推正是取 $m = 3$(Yoshida--Suzuki 三跳法)或 $m = 5$(5 级公式)
    的解 $s_1 = s_3 = \frac{1}{2 - 2^{1/(p+1)}}$ 与
    $s_1 = s_2 = s_4 = s_5 = \frac{1}{4 - 4^{1/(p+1)}}$ 迭代应用此引理的结果.
\end{Proof}

\begin{Note}[阶数与代价的权衡]
    高阶公式改进了对 $t$ 和 $\varepsilon$ 的依赖($1/p$ 指数), 但每步的指数个数随阶数
    指数增长($5^{k-1}$), 实践中 2 至 6 阶往往最优. 常系数 $s_k$ 的构造
    (Yoshida--Suzuki 三跳法与 5 级 Trotter--Suzuki 法)是数值分析中组合方法的直接移植.
\end{Note}

\begin{Note}[与经典算子分裂与谱方法的对比]
    Trotter--Suzuki 公式并非量子特有: 它们是数值分析中\textbf{算子分裂
    (operator splitting)}与\textbf{组合方法(composition methods)}的量子实现.
    一阶 Lie--Trotter 分裂与二阶 \textbf{Strang 分裂(Strang splitting)}(1968)是经典求解
    对流扩散型 PDE 的标准工具; 高阶组合方法及其系数($s_k$)由 Yoshida(1990)与
    Suzuki(1991)在量子计算之前就建立. 二者的区别只在载体:
    \begin{itemize}
        \item \textbf{经典分裂}: 矩阵作用在网格向量上, 每步 $O(N)$ 次浮点运算
        ($N$ 为网格点数); 谱方法则先用 FFT 对角化微分算子, 每步
        $O(N\log N)$ 次运算, 再显式读出谱系数组装解;
        \item \textbf{量子版本}: 酉演化作用在 $n = \log N$ 个量子比特上, 
        门数 $O(\mathrm{poly}(n))$; 当初始态是叠加态, 且只需提取低维观测值
        (期望, 关联函数)而不需完整解向量时, 量子版本在高维
        ($d \gtrsim 5$)下可获得对网格数的\textbf{指数加速(exponential speedup)}.
    \end{itemize}
    这解释了谱方法/分裂法在经典 PDE 数值解中的角色: 谱方法用 FFT 把微分算子
    对角化, 与量子 qubitization 把 $H$ 变为旋转矩阵的思路同构 --- 区别在于
    经典需要显式对角化矩阵, 量子只需要 block-encoding 的"隐式对角化".
\end{Note}

\subsection{局部演化的实现与交换情形}

乘积公式的每个因子 $e^{-i\tau H_j}$ 仍需分解为基本门. 两种常见情形:

\begin{Eg}[交换哈密顿量的精确模拟]
    若 $H = \sum_j H_j$ 且各 $H_j$ 两两对易, 则 $e^{-iHt} = \prod_j e^{-iH_j t}$ 精确成立, 
    无需任何分裂误差. 例如 TFIM 的 $H_1 = \sum_i Z_i Z_{i+1}$ 与
    $H_2 = \sum_i X_i$ 各自内部对易.
\end{Eg}

\begin{Eg}[两比特 Pauli 项的电路]
    对 $H = ZZ$(省略常系数), 由 $X e^{-itZ} X = e^{itZ}$ 得
    \begin{equation*}
    \Qcircuit @C=1em @R=.8em {
        \lstick{\ket{x}} & \ctrl{1} & \qw & \ctrl{1} & \rstick{\ket{x}} \qw \\
        \lstick{\ket{y}} & \targ & \gate{R_z(2t)} & \targ & \rstick{\ket{y}} \qw
    }
    \end{equation*}
    即 $e^{-it\,Z\otimes Z} = \mathrm{CNOT}\cdot\big(I \otimes R_z(2t)\big)\cdot\mathrm{CNOT}$.
    一般 $k$ 比特 Pauli 项可先用 CNOT 链计算奇偶性, 施加单个 $R_z$ 后再用 CNOT 链还原
    (共 $2(k-1)$ 个 CNOT 与 1 个 $R_z$); 含 $X$, $Y$ 的项则先用 Clifford 共轭
    ($X = HZH$, $Y = SHS^{\dagger} Z S H S^{\dagger}$ 类关系)对角化后再处理.
\end{Eg}

\subsection{截断 Taylor 级数: LCU 方法}

乘积公式把 $e^{-iHt}$ 分解为一系列可精确实现的短时演化, 但误差对精度的依赖只有
$O(\varepsilon^{-1/p})$(多项式). 对一般哈密顿量, 由\textbf{无快速前推(no-fast-forwarding)}
下界, 黑盒模拟 $e^{-iHt}$ 至少需要 $\Omega(t)$ 次查询, 而乘积公式的上界关于 $t$
是超线性的($O(t^{1+1/p})$). 一个能达到(几乎)最优复杂度的方案是: 直接把 $e^{-iHt}$
做\textbf{Taylor 级数展开}, 把截断结果表示为\textbf{线性酉组合(linear combination of unitaries, LCU)},
再用上一节的\textbf{非感知振幅放大(OAA)}把后选择成功概率恢复到常数. 这就是
\textbf{截断 Taylor 级数方法}(Berry--Childs--Cleve--Kothari--Somma, 2015).

\begin{Def}[哈密顿量的 LCU 分解]
    把哈密顿量写成酉算子的非负线性组合
    \begin{equation}
        H = \sum_{l=1}^{L} \alpha_l\, U_l, \qquad \alpha_l > 0, \quad U_l \text{ 为酉算子},
    \end{equation}
    并记其\textbf{1-范数(one-norm)}
    \begin{equation}
        \lambda := \sum_{l=1}^{L} \alpha_l.
    \end{equation}
    例如对 Pauli 分解 $H = \sum_j c_j P_j$, 取 $\alpha_l = |c_l|$,
    $U_l = \mathrm{sign}(c_l)\, P_l$; 因 $P_l$ 是酉的 Pauli 算符, $U_l$ 仍是酉算子
    (符号只是整体相位). 由三角不等式,
    \begin{equation}
        \norm{H} = \norm{\sum_l \alpha_l U_l} \le \sum_l \alpha_l = \lambda.
    \end{equation}
\end{Def}

\begin{Note}[与乘积公式的对比]
    Trotter 要求 $H$ 是少量项之和, 且每项指数 $e^{-i\tau H_j}$ 可精确实现;
    LCU 分解只要求每个 $U_l$ 可被(受控地)实现, 对任意 $\mathrm{poly}(n)$ 项的
    Pauli 哈密顿量都成立, 代价则表现为对 1-范数 $\lambda$ 的依赖. 注意 $\lambda$
    可能远大于 $\norm{H}$: 例如 $H = \sum_{j=1}^{L} X_j$ 时 $\norm{H} = 1$ 而
    $\lambda = L$, 因此该方法对"系数和很大"的哈密顿量并不经济; 更紧的构造
    (qubitization)可把依赖改进到 $\norm{H}$(见下一模块).
\end{Note}

\begin{Def}[截断 Taylor 级数]
    把总时间 $t$ 分为 $r$ 段, 每段 $\tau = t/r$. 对单段做 Taylor 展开并截断到 $K$ 阶:
    \begin{equation}
        e^{-iH\tau} = \sum_{k=0}^{\infty} \frac{(-iH\tau)^k}{k!}
        \;\approx\; \widetilde U_K(\tau) := \sum_{k=0}^{K} \frac{(-iH\tau)^k}{k!}.
    \end{equation}
    代入 $H^k = \sum_{l_1,\dots,l_k}\alpha_{l_1}\cdots\alpha_{l_k}\, U_{l_1}\cdots U_{l_k}$
    (酉算子的乘积仍是酉算子):
    \begin{equation}
        \widetilde U_K(\tau) = \sum_{k=0}^{K}\;\sum_{l_1,\dots,l_k}
        \frac{(-i\tau)^k}{k!}\,\alpha_{l_1}\cdots\alpha_{l_k}\,
        U_{l_1}\cdots U_{l_k}.
    \end{equation}
    去掉模 $1$ 的相位 $(-i)^k$, 这是酉算子 $\{U_{l_1}\cdots U_{l_k}\}$ 的一个 LCU, 
    其 1-范数(归一化常数)为
    \begin{equation}
        s := \sum_{k=0}^{K}\frac{\tau^k}{k!}\sum_{l_1,\dots,l_k}\alpha_{l_1}\cdots\alpha_{l_k}
        = \sum_{k=0}^{K}\frac{(\lambda\tau)^k}{k!} \;\le\; e^{\lambda\tau}.
    \end{equation}
\end{Def}

\begin{Lem}[LCU 引理]
    设 $M = \sum_j w_j V_j$($w_j \ge 0$, $V_j$ 酉), $s = \sum_j w_j$. 构造
    \textbf{PREPARE} $V: \ket{0^a}\mapsto \sum_j \sqrt{w_j/s}\,\ket{j}$ 与
    \textbf{SELECT} $\mathrm{SEL} = \sum_j \ket{j}\bra{j}\otimes V_j$, 则
    \begin{equation}
        (V^\dagger\otimes I)\,\mathrm{SEL}\,(V\otimes I)\ket{0^a}\ket{\psi}
        = \ket{0^a}\otimes \frac{1}{s}M\ket{\psi} \;+\; \ket{0^a}^{\perp}\otimes(\cdots),
    \end{equation}
    其中 $\ket{0^a}^{\perp}$ 表示辅助寄存器中与 $\ket{0^a}$ 正交的分量.
    测量辅助寄存器得到 $\ket{0^a}$ 的概率为 $\norm{M\ket{\psi}}^2/s^2$,
    成功时系统态恰为 $M\ket{\psi}/\norm{M\ket{\psi}}$.
\end{Lem}

\begin{Proof}
    三步逐项追踪:
    \begin{equation*}
    \begin{aligned}
        \ket{0^a}\ket{\psi}
        &\xrightarrow{\,V\otimes I\,} \sum_j \sqrt{\frac{w_j}{s}}\,\ket{j}\ket{\psi}
        \xrightarrow{\;\mathrm{SEL}\;} \sum_j \sqrt{\frac{w_j}{s}}\,\ket{j}\,V_j\ket{\psi} \\
        &\xrightarrow{\,V^\dagger\otimes I\,}
        \sum_j \sqrt{\frac{w_j}{s}}\,\big(V^\dagger\ket{j}\big)V_j\ket{\psi}.
    \end{aligned}
    \end{equation*}
    由 $V^\dagger\ket{j}$ 在 $\ket{0^a}$ 上的分量 $\braket{0^a}{V^\dagger j}
    = \braket{V 0^a}{j} = \sqrt{w_j/s}$, 提取 $\ket{0^a}$ 分量得
    \[ \ket{0^a}\otimes\sum_j\frac{w_j}{s}V_j\ket{\psi}
       = \ket{0^a}\otimes\frac{1}{s}M\ket{\psi}, \]
    余下为与 $\ket{0^a}$ 正交的分量. 由 Born 规则即得成功概率与成功态.
\end{Proof}

\begin{Note}[截断误差与截断阶的选取]
    由 $\norm{H}\le\lambda$ 与 Taylor 余项,
    \begin{equation}
        \norm{e^{-iH\tau}-\widetilde U_K(\tau)}
        \le \sum_{k>K}\frac{\norm{H}^k\tau^k}{k!}
        \le \sum_{k>K}\frac{(\lambda\tau)^k}{k!}.
    \end{equation}
    对 $0\le x\le1$, 用 $(K+2)\cdots(K+1+m)\ge m!$ 放缩:
    \begin{equation}
        \sum_{k>K}\frac{x^k}{k!} = \frac{x^{K+1}}{(K+1)!}\sum_{m\ge0}\frac{(K+1)!}{(K+1+m)!}
        \le \frac{x^{K+1}}{(K+1)!}\,e^{x}
        \le e\left(\frac{e}{K+1}\right)^{K+1},
    \end{equation}
    其中最后一步用了 $x\le1$ 与 Stirling 下界 $(K+1)!\ge\big(\frac{K+1}{e}\big)^{K+1}$.
    因此要使尾项 $\le\delta$, 只需
    \[ K = \Theta\Big(\frac{\log(1/\delta)}{\log\log(1/\delta)}\Big), \]
    即截断阶仅\textbf{对数}(而非幂次)依赖于精度.
\end{Note}

\begin{Thm}[截断 Taylor 级数方法的复杂度]
    设 $H = \sum_l\alpha_l U_l$, $\lambda = \sum_l\alpha_l$. 取 $r = \lceil\lambda t\rceil$ 段
    (使 $\lambda\tau = \lambda t/r \le 1$), 每段截断阶
    \begin{equation}
        K = \Theta\Big(\frac{\log(\lambda t/\varepsilon)}{\log\log(\lambda t/\varepsilon)}\Big),
    \end{equation}
    使单段截断误差 $\le \varepsilon/r$. 则每段 LCU 的归一化常数 $s \le e^{\lambda\tau}\le e$,
    $r$ 段误差线性累积后总误差 $\le \varepsilon$; 用 OAA 把每段后选择成功概率提升到常数,
    总查询复杂度为
    \begin{equation}
        \boxed{O\!\Big(\lambda t \cdot \frac{\log(\lambda t/\varepsilon)}{\log\log(\lambda t/\varepsilon)}\Big)}.
    \end{equation}
\end{Thm}

\begin{Proof}[复杂度核算]
    每段由三个代价构成: (1) \textbf{PREPARE/SELECT} 把截断级数 $\widetilde U_K(\tau)$
    制备为 LCU, SELECT 一次施加至多 $K$ 个 $U_l$(即 $O(K)$ 次基本 oracle 查询);
    (2) 该 LCU 的成功概率约为 $\norm{\widetilde U_K(\tau)\ket{\psi}}^2/s^2$,
    因 $\widetilde U_K(\tau)\approx e^{-iH\tau}$ 近酉, 成功概率 $\approx 1/s^2=\Omega(1)$;
    (3) 由上一节的非感知振幅放大, $O(s)=O(1)$ 次迭代即可把成功概率提升到常数.
    于是每段总代价 $O(sK)=O(K)$, $r$ 段合计
    \begin{equation}
        O(rK) = O\!\Big(\lambda t\,\frac{\log(\lambda t/\varepsilon)}{\log\log(\lambda t/\varepsilon)}\Big).
    \end{equation}
    常数因子与隐式系数见 Berry 等(2015)原文, 这里只关心对 $\lambda, t, \varepsilon$ 的渐近依赖.
\end{Proof}

\begin{Note}[最优性与对下一模块的展望]
    该复杂度关于 $t$ 是\textbf{线性的}, 关于 $1/\varepsilon$ 是\textbf{多重对数}的, 
    除 $\log\log$ 因子外已达到无快速前推下界 $\Omega(t)$; 对比乘积公式
    $O(t^{1+1/p}/\varepsilon^{1/p})$(关于 $\varepsilon$ 是幂次的), 截断 Taylor 级数
    在 $\varepsilon\to0$ 时显著更优. 其"先做 LCU(block-encoding)再绕过了后选择代价"的
    结构, 正是下一模块 \textbf{qubitization} 与 \textbf{QSVT} 的雏形: qubitization 把
    复杂度进一步改进为 $O(\lambda t + \log(1/\varepsilon))$(消去 $\log\log$ 因子),
    QSVT 则把同样的机制推广到任意矩阵函数 $f(H)$(见 QPE 一节对矩阵函数的讨论).
\end{Note}

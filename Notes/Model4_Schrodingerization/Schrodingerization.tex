\section{引言: 从线性代数到微分方程}

前三个模块建立了量子算法从"语言"到"构件"再到"统一框架"的完整链条:
Module1 给出了量子计算的语言(\textbf{量子比特(qubit)}、\textbf{测量(measurement)}、\textbf{门(gate)}), Module2 介绍了 \textbf{量子傅里叶变换(Quantum Fourier Transform, QFT)}、\textbf{量子相位估计(Quantum Phase Estimation, QPE)}、\textbf{振幅放大(amplitude amplification)}与
Trotter 型\textbf{哈密顿量模拟(Hamiltonian simulation)}四类原语, Module3 用 \textbf{块编码(block-encoding)}、\textbf{量子比特化(qubitization)} 与 \textbf{量子信号处理(Quantum Signal Processing, QSP)}/\textbf{量子奇异值变换(Quantum Singular Value Transformation, QSVT)}
把它们统一到"矩阵函数变换"的框架之下. 本模块完成最后一块拼图: 把这些构件用于
\textbf{科学计算}, 特别是\textbf{偏微分方程(Partial Differential Equation, PDE)}与线性代数的量子求解.

\begin{Note}[两条主线]
    \begin{enumerate}
        \item \textbf{HHL 算法(Harrow--Hassidim--Lloyd)}(Session 8): 用量子相位估计求解\textbf{线性方程组(system of linear equations)} $Ax = b$,
        是"微分方程 $\to$ 线性代数 $\to$ 量子线路"路线的代表;
        \item \textbf{薛定谔化方法(Schrödingerisation)}(Session 9): 把一般线性 \textbf{常微分方程(Ordinary Differential Equation, ODE)}/PDE 通过
        \textbf{扭曲相位变换(warped phase transformation)} 转化为\textbf{薛定谔方程(Schrödinger equation)}, 从而直接套用
        哈密顿量模拟 —— 这是 PDE 量子模拟的现代通用框架.
    \end{enumerate}
    两条主线共享同一套底层构件(block-encoding、哈密顿量模拟、振幅放大),
    因此在进入正题之前, 本模块先对几类常用哈密顿量模拟方法做一次综述, 作为两者的共同基础.
\end{Note}

本模块的结构如下: §2 介绍 HHL 算法(线性代数路线的代表, 含其现代替代 QSVT-HHL);
§3 综述四类常用哈密顿量模拟方法(Trotter、Taylor+LCU、qubitization/QSVT), 作为共同构件;
§4 是核心, 系统展开薛定谔化方法(warped phase transformation、恢复定理、
smooth initialization、带源项与时间依赖系统、复杂度与量子优势、病态 PDE);
§5 把薛定谔化放在酉化方法族(LCHS、Transmutation、moment-matching dilation)的统一视角下比较;
§6 汇总文献与延伸. 记号上, $\ket{\cdot}$ 表示量子态, $\norm{\cdot}$ 表示算子范数(对向量为欧氏范数),
$A^\dagger$ 表示共轭转置.

\section{HHL 算法: 求解线性方程组}

\subsection{问题设定}

\begin{Def}[\textbf{量子线性方程组问题(Quantum Linear System Problem, QLSP)}]
    给定 $n$ 比特矩阵 $A \in \mathbb{C}^{N\times N}$($N = 2^n$)与归一化右端向量
    $\ket{b} = U_b \ket{0^n}$, 求解
    \begin{equation}
        A x = b.
    \end{equation}
    由于解 $x = A^{-1}b$ 一般不归一化, 目标不是恢复 $x$ 本身, 而是制备\textbf{归一化解态}
    \begin{equation}
        \ket{x} = \frac{A^{-1}\ket{b}}{\norm{A^{-1}\ket{b}}},
    \end{equation}
    并使其与真实解满足 $\norm{\ket{x} - \ket{\bar{x}}} \le \varepsilon$.
\end{Def}

\begin{Note}[厄米假设与膨胀]
    不失一般性可假设 $A$ 是\textbf{厄米矩阵(Hermitian matrix)}: 否则求解其\textbf{厄米膨胀(Hermitian dilation)}
    \begin{equation}
        \bar{A} = \begin{pmatrix} 0 & A \\ A^\dagger & 0 \end{pmatrix}
        = \ket{1}\bra{0}\otimes A + \ket{0}\bra{1}\otimes A^\dagger,
        \qquad
        \bar{A}\ket{0}\ket{x} = \ket{1}\ket{b},
    \end{equation}
    把矩阵维数扩大一倍(增加一个\textbf{辅助比特(ancilla)})即可. 此外通过归一化 $A/\norm{A}$ 可进一步假设
    \begin{equation}
        \norm{A} = 1, \qquad 0 < \lambda_0 \le \lambda_1 \le \dots \le \lambda_{N-1} = 1,
    \end{equation}
    于是\textbf{条件数(condition number)}
    \begin{equation}
        \kappa := \norm{A}\,\norm{A^{-1}} = \norm{A^{-1}} = \frac{1}{\lambda_0}.
    \end{equation}
    归一化 $A \to A/\norm{A}$ 只改变解的整体尺度(归一化解态 $\ket{x}$ 不变), 故无损一般性.
    以下始终假设 $\norm{A} = 1$.
\end{Note}

\subsection{算法描述}

设 $A = \sum_j \lambda_j \ket{v_j}\bra{v_j}$ 为\textbf{谱分解(spectral decomposition)}, $\ket{b} = \sum_j \beta_j \ket{v_j}$.
解可以写成 $A^{-1}\ket{b} = \sum_j \beta_j \lambda_j^{-1}\ket{v_j}$, 因此算法只需
"读出"每个\textbf{本征值(eigenvalue)} $\lambda_j$ 并乘上系数 $\lambda_j^{-1}$ —— 这正是 QPE 擅长的任务.

\begin{Def}[HHL 三步骤]
    记 $U = e^{i2\pi A}$(由哈密顿量模拟实现), HHL 算法由三步组成:
    \begin{enumerate}
        \item \textbf{QPE}: 施加 $U_{\mathrm{QPE}}$ 于 $d$ 比特\textbf{相位寄存器(phase register)}与\textbf{系统寄存器(system register)},
        把本征值编码到相位寄存器:
        \begin{equation}
            U_{\mathrm{QPE}} \ket{0^d}\ket{v_j} = \ket{\lambda_j}\ket{v_j},
            \qquad
            U_{\mathrm{QPE}} \ket{0^d}\ket{b} = \sum_j \beta_j \ket{\lambda_j}\ket{v_j};
        \end{equation}
        \item \textbf{受控旋转(controlled rotation)}: 以相位寄存器为控制, 对\textbf{信号比特(signal qubit)}施加旋转
        \begin{equation}
            U_{\mathrm{CR}} \ket{0}\ket{\lambda_j}
            = \Big(\sqrt{1 - C^2/\lambda_j^2}\,\ket{0} + \frac{C}{\lambda_j}\ket{1}\Big)\ket{\lambda_j},
        \end{equation}
        其中常数 $C$ 满足 $0 < C \le \lambda_0$(见 \ref{subsec:hhl-cr} 节);
        \item \textbf{逆 QPE(inverse QPE, uncomputation)}: 施加 $U_{\mathrm{QPE}}^\dagger$ 把相位寄存器
        复位为 $\ket{0^d}$, 使之成为可复用的工作寄存器.
    \end{enumerate}
    整体记为 $U_{\mathrm{HHL}}$, 电路如下:
    \begin{equation*}
    \Qcircuit @C=1em @R=.8em {
        \lstick{\ket{0}} & \qw & \qw & \ctrl{1} & \qw & \qw & \meter & \rstick{0/1} \qw \\
        \lstick{\ket{0^d}} & \gate{U_{\mathrm{QPE}}} & \gate{U_{\mathrm{CR}}} & \gate{U_{\mathrm{QPE}}^\dagger} & \qw & \qw & \qw & \qw \\
        \lstick{\ket{b}} & \qw & \qw & \qw & \qw & \qw & \qw & \rstick{\ket{x}} \qw
    }
    \end{equation*}
    其中 $U_{\mathrm{QPE}}$ 同时(受控地)作用于系统寄存器, 系统寄存器穿过其方框;
    $U_{\mathrm{CR}}$ 表示以相位寄存器(内容为 $\lambda_j$)为控制的旋转
    (电路图中的控制线示意其控制关系, 实际的受控目标在相位寄存器上).
\end{Def}

\begin{Thm}[HHL 的输出]
    上述电路实现
    \begin{equation}
        U_{\mathrm{HHL}} \ket{0}\ket{b}
        = \sum_j \beta_j \Big(\sqrt{1 - C^2/\lambda_j^2}\,\ket{0}
        + \frac{C}{\lambda_j}\ket{1}\Big)\ket{v_j}.
    \end{equation}
    测量信号比特: 得到 $1$ 时系统寄存器处于(归一化的)
    \begin{equation}
        \ket{x} \propto \sum_j \beta_j \frac{C}{\lambda_j}\ket{v_j} = C A^{-1}\ket{b},
    \end{equation}
    即所求近似解(常数 $C$ 在归一化后消失).
\end{Thm}

\begin{Note}[恢复解的范数]
    成功概率 $p(1) = \norm{C A^{-1}\ket{b}}^2$, 因此通过估计 $p(1)$
    (重复运行电路)可以恢复 $\norm{A^{-1}\ket{b}}$.
\end{Note}

\subsection{受控旋转的实现} \label{subsec:hhl-cr}

\begin{Prop}[按旋转角的受控旋转]
    设 $0 \le \theta < 1$ 有 $d$ 位\textbf{定点表示(fixed-point representation)} $\theta = (0.\theta_{d-1}\cdots\theta_0)$.
    由 $e^{-i\theta Y}$ 的分解, 旋转角 $\theta$ 的受控旋转可写为
    \begin{equation}
        U = \sum_{j \in [2^d]} \exp\big(-i (0.j_{d-1}\cdots j_0) Y\big) \otimes \ket{j}\bra{j},
    \end{equation}
    即一串以单个比特为控制的 $Y$ 旋转 $R_y(\theta/2^{k})$, 共 $O(d)$ 个门.
\end{Prop}

\begin{Note}[$U_{\mathrm{CR}}$ 的完整构造]
    为把本征值 $\lambda_j$(相位寄存器中的 $d'$ 位定点数)转化为旋转角, 取
    \begin{equation}
        \theta_j = \arcsin(C/\lambda_j),
    \end{equation}
    用经典算术电路实现映射 $U_{\mathrm{angle}}: \ket{\lambda_j} \mapsto \ket{\theta_j}$
    (需要 $O(\mathrm{poly}(d'))$ 个门与额外工作寄存器), 施加旋转后再用
    $U_{\mathrm{angle}}^\dagger$ 做 uncomputation. 由此
    \[ \sin\theta_j = C/\lambda_j, \qquad
       U_{\mathrm{CR}}\ket{0}\ket{\lambda_j}
       = \big(\cos\theta_j\ket{0} + \sin\theta_j\ket{1}\big)\ket{\lambda_j}. \]
    精度要求: 由 §2.4 的分析, 需把 $\lambda_j$ 估计到加性精度 $O(\varepsilon/\kappa)$,
    故相位寄存器取 $d' = O(\log(\kappa/\varepsilon))$ 位; 由 Module1 的误差线性增长
    论证, $U_{\mathrm{angle}}$ 的算术实现误差与 $\lambda_j$ 的 QPE 误差同阶即可,
    不构成额外主导项.
\end{Note}

\subsection{复杂度分析}

\begin{Thm}[HHL 的\textbf{查询复杂度(query complexity)}]
    设 $A$ 厄米正定且 $\norm{A} = 1$, $\lambda_0 = 1/\kappa$, 取 $C = \lambda_0 = 1/\kappa$ 最大化成功概率. 则
    \begin{enumerate}
        \item 成功概率: 最坏情形
        \begin{equation}
            p(1) = \norm{C A^{-1}\ket{b}}^2 \ge \frac{C^2}{\norm{A}^2} = C^2 = \Omega(\kappa^{-2}),
        \end{equation}
        其中第一个不等式来自 $\norm{b} = \norm{AA^{-1}b} \le \norm{A}\,\norm{A^{-1}b}$
        (即 $\norm{A^{-1}b} \ge \norm{b}/\norm{A} = 1$, 因 $\norm{A} = 1$);
        \item 精度: 为把归一化解估计到相对精度 $\varepsilon$, QPE 需要对本征值 $\lambda_j$ 的
        \textbf{加性精度} $\varepsilon/\kappa$(因为对 $\lambda_j$ 的估计误差
        $\delta_j$ 使 $\lambda_j^{-1}$ 产生相对误差 $O(\delta_j/\lambda_j) \le O(\varepsilon)$),
        因此单次运行对 $U = e^{i2\pi A}$ 的查询数为 $O(\kappa/\varepsilon)$;
        \item 总复杂度: 成功概率 $\Omega(\kappa^{-2})$ 意味着直接重复需 $O(\kappa^2)$ 次(AA 之前),
        得最坏情形 $O(\kappa^3/\varepsilon)$; 用振幅放大(见 Module2)把重复次数降为
        $O(\kappa)$, 得
        \begin{equation}
            \text{HHL 查询复杂度} = O\!\Big(\frac{\kappa^2}{\varepsilon}\Big)
            \quad(\text{含 AA}), \qquad
            O\!\Big(\frac{\kappa}{\varepsilon}\Big)
            \quad(\text{最佳情形: } \norm{A^{-1}\ket{b}} = \Omega(1)).
        \end{equation}
    \end{enumerate}
\end{Thm}

\begin{Proof}[思路]
    AA 的适用性: 由 $U_{\mathrm{HHL}}\ket{0}\ket{b}$ 的形式,
    好态由单个信号比特 $\ket{1}$ 标志, 反射算子即 $R_{\mathrm{good}} = Z \otimes I$;
    关于初态的反射由 $U_0 = U_{\mathrm{HHL}}(I \otimes U_b)$ 构造.
    最坏情形的 $O(\kappa^3/\varepsilon)$ 到 $O(\kappa^2/\varepsilon)$ 的改进正是
    振幅放大把 $O(\kappa^2)$ 次重复降为 $O(\kappa)$ 次的结果.
\end{Proof}

\subsection{HHL 的现代替代: QSVT-HHL}

\begin{Note}[QPE 路线与 QSVT 路线的对比]
    HHL 的精度依赖 $1/\varepsilon$ 来自 QPE 的位数; Module3 的 QSVT 直接
    用多项式逼近 $f(x) = 1/(\kappa x)$, 无需相位寄存器, 也无须逐比特读出本征值:
    \begin{enumerate}
        \item \textbf{多项式度}: 在区间 $[-1,-1/\kappa]\cup[1/\kappa,1]$ 上逼近
        $1/(\kappa x)$ 的奇多项式度数为 $O(\kappa\log(\kappa/\varepsilon))$;
        \item \textbf{查询复杂度}: 对 $U_A$ 及其逆为
        $O\!\big(\frac{\kappa^2}{\zeta}\log(\kappa/\varepsilon)\big)$($\zeta=\norm{A^{-1}\ket{b}}\ge1$，故最坏情形 $O(\kappa^2\log(\kappa/\varepsilon))$; 当 $\ket{b}$ 与最小奇异方向有 $\Omega(1)$ 重叠时 $\zeta=\Omega(\kappa)$、降为 $O(\kappa\log(\kappa/\varepsilon))$);
        \item \textbf{精度依赖}: $\log(1/\varepsilon)$ 代替 $1/\varepsilon$;
        \item \textbf{无需膨胀}: QSVT 直接处理\textbf{非厄米矩阵(non-Hermitian matrix)}(奇异值变换),
        HHL 的 QPE 路线要求 $A$ 厄米(否则先膨胀).
    \end{enumerate}
    两种路线的对比总结如下(设系统规模 $N = 2^n$, 忽略 $\mathrm{poly}\log$ 因子):
    \begin{center}
    \begin{tabular}{ccc}
        \toprule
        方法 & 查询复杂度(主导项) & 条件 \\
        \midrule
        HHL(QPE) & $\tilde O\big(\kappa^2/\varepsilon\big)$ & 需厄米 + 哈密顿量模拟 \\
        QSVT-HHL & $\tilde O\big(\kappa^2\log(\kappa/\varepsilon)\big)$ & 只需 block-encoding \\
        \bottomrule
    \end{tabular}
    \end{center}
    核心差异: QSVT 路线以 $\log(1/\varepsilon)$ 代替 $1/\varepsilon$(精度依赖),
    且无需厄米假设(直接做奇异值变换, 免去膨胀).
\end{Note}

\begin{Note}[与经典线性方程组解法的对比]
    经典的线性方程组解法复杂度以 $N$($A$ 的维数)、稀疏度与条件数 $\kappa$ 衡量:
    \begin{center}
    \begin{tabular}{@{}>{\raggedright\arraybackslash}p{5.2cm}>{\raggedright\arraybackslash}p{4.4cm}>{\raggedright\arraybackslash}p{4.2cm}@{}}
        \toprule
        方法 & 复杂度 & 说明 \\
        \midrule
        \textbf{高斯消元 / LU(Gaussian elimination / LU decomposition)} & $O(N^3)$ & 稠密矩阵, 直接法 \\
        \textbf{共轭梯度(Conjugate Gradient, CG)} & $O(N\sqrt{\kappa}\log(1/\varepsilon))$ & SPD 且稀疏 \\
        \textbf{广义最小残差法(Generalized Minimal Residual, GMRES)} 等 \textbf{Krylov 法(Krylov subspace method)} & 与谱分布有关 & 一般稀疏, 每步 $O(\mathrm{nnz}(A))$ \\
        HHL (QPE 路线) & $O(\mathrm{poly}(\log N)\,\kappa^2/\varepsilon)$ & 需 $e^{i2\pi A}$ 可模拟 \\
        QSVT-HHL & $O(\mathrm{poly}(\log N)\,\kappa^2\log(\kappa/\varepsilon))$ & 需 block-encoding \\
        \bottomrule
    \end{tabular}
    \end{center}
    HHL 对 $N$ 的依赖是 $\mathrm{poly}(\log N)$ —— 对经典方法($O(N)$ 或 $O(N^3)$)
    是指数级改进. 但这个优势有三个前提, 也是本笔记反复强调的"诚实性"要求:
    (1) $e^{i2\pi A}$(或 $U_A$)必须能由 $\mathrm{poly}(n)$ 个门实现 —— 对\textbf{稀疏矩阵(sparse matrix)}
    由哈密顿量模拟保证, 对一般\textbf{稠密矩阵(dense matrix)}不成立;
    (2) 输出是量子态 $\ket{x}$, 直接读出整个解向量需 $\Omega(N)$ 次采样, 因此
    HHL 只在"解作为更大算法的子程序"或"只需低维观测值 $\bra{x}O\ket{x}$"时
    才有优势(后者仍需 $O(1/\varepsilon^2)$ 次采样);
    (3) 对 $\kappa$ 的依赖(线性或二次)远差于\textbf{经典迭代法(classical iterative method)} —— 若 $A$ 病态,
    HHL 的 $\kappa$ 开销可能抵消规模上的指数优势.
\end{Note}

\begin{Note}[HHL 的讨论]
    (1) \textbf{查询下界(query lower bound)}: 一般 QLSP 求解器至少需要 $\Omega(\kappa)$ 次查询
    (可由多项式方法/相位估计型的下界导出); (2) \textbf{精度}: 上面表格中的 $\varepsilon$
    为解的相对精度; (3) \textbf{与经典迭代法的关键差异}: 经典 CG 每次迭代
    $O(\mathrm{nnz}(A))$ 次浮点运算但需要显式访问 $A$ 的元素,
    HHL 每次查询 $A$ 的代价是 $\mathrm{poly}(n)$ 的门 —— 二者的代价模型
    (浮点运算 vs 量子门)不同, 直接的常数对比没有意义, 只有渐进阶才有意义.
\end{Note}

\section{常用哈密顿量模拟方法综述}

哈密顿量模拟是 HHL 与薛定谔化的共同底层构件, 本模块先对前两个模块已建立的方法做统一梳理,
并补充两种基于泰勒级数/随机化的方法, 便于后续对比. 本节同时给出各方法的\textbf{严格误差界}
与\textbf{查询下界}, 使复杂度比较建立在可核验的数学基础之上.

\subsection{问题、任务模型与输入模型}

\begin{Def}[哈密顿量模拟问题]
    给定厄米矩阵 $H = H^\dagger \in \mathbb{C}^{N\times N}$($N = 2^n$), 制备
    $\ket{\psi(t)} = e^{-iHt}\ket{\psi_0}$ 到精度 $\varepsilon$: 即构造线路
    $\widetilde U(t)$ 使
    \begin{equation}
        \norm{\widetilde U(t) - e^{-iHt}} \le \varepsilon
    \end{equation}
    (算子范数意义, 与初态无关的\textbf{态到态(state-to-state)}精度).
\end{Def}

\begin{Note}[任务模型: 计入与不计入的成本]
    按 day17 课件的约定, 复杂度分析计入: 对 $U_H$(或稀疏 oracle)的\textbf{查询次数}、
    逻辑门数与线路深度、postselection/振幅放大的重复调用;
    \textbf{不计入}: 初态制备、完整经典读出、物理层纠错开销.
    区分"查询复杂度"与"门复杂度"很重要: 前者隐藏了 $U_H$ 的实现代价,
    后者才反映真实线路规模; 两者可能相差巨大(如 $U_H$ 本身需要深度线路时).
\end{Note}

\begin{Note}[三种输入模型]
    量子计算机访问矩阵 $H$ 的方式(输入模型)决定复杂度的量纲:
    \begin{enumerate}
        \item \textbf{稀疏 oracle 模型}: 设 $H$ 是 $s$-稀疏的(每行每列至多 $s$ 个非零元),
        可查询\textbf{列/行索引 oracle} $O_c\ket j\ket\ell = \ket j\ket{c(j,\ell)}$($c(j,\ell)$ 为第 $j$ 行第 $\ell$ 个非零元所在列)
        与\textbf{矩阵元 oracle} $O_A\ket j\ket k\ket 0 = \ket j\ket k\ket{H_{jk}}$;
        \item \textbf{block-encoding 模型}(Module3): $H$ 的 $(\alpha, a, \varepsilon)$-block-encoding
        $U_H$, 查询数为对 $U_H, U_H^\dagger$ 的调用次数, 归一化因子 $\alpha \ge \norm{H}$;
        \item \textbf{Pauli 分解模型}: $H = \sum_{j} c_j P_j$($P_j$ 为 Pauli 串),
        常用于 Trotter 方法, 代价以对 $e^{-i\tau c_j P_j}$ 的调用次数衡量.
    \end{enumerate}
    对 PDE 离散矩阵, 三者在"如何把 $A_h$ 变成线路可访问对象"这一问题上等价于
    Module3 的 block-encoding 构造; 本节以下以 $U_H$ 的查询数为统一单位,
    并注明各方法所需的额外辅助比特.
\end{Note}

\begin{Note}[无快速前推定理(no-fast-forwarding)]
    对通过黑盒 oracle 访问的\textbf{一般}稀疏哈密顿量 $H$,
    模拟 $e^{-iHt}$ 至少需要
    \begin{equation}
        \Omega(t)
    \end{equation}
    次查询(Berry--Ahokas--Cleve--Sanders 2007). 直觉: 每个 oracle 查询至多贡献 $O(1)$ 的
    "演化量", 一般矩阵的谱信息无法被压缩. 该下界不排除\textbf{特殊结构}的快速前推
    (如 $U = R_z(\varphi)$ 时 $U^j$ 的代价与 $j$ 无关), 但这类结构在科学计算的一般矩阵中很少见.
    结合精度的下界 $\Omega\big(\log(1/\varepsilon)/\log\log(1/\varepsilon)\big)$
    (见 §3.4), 哈密顿量模拟的查询下界为
    \begin{equation}
        \Omega\!\Big(\alpha t + \frac{\log(1/\varepsilon)}{\log\log(1/\varepsilon)}\Big),
    \end{equation}
    其中 $\alpha$ 为 block-encoding 的归一化因子(稀疏模型下取 $\alpha = \norm{H}$).
\end{Note}

\subsection{Trotter/Suzuki 乘积公式}

回顾 Module2: 对 $H = \sum_{j=1}^m H_j$, 一阶 \textbf{Lie--Trotter 公式}
$S_1(\tau) = \prod_{j=1}^m e^{-i\tau H_j}$, 二阶对称公式
$S_2(\tau) = \Big(\prod_{j=1}^m e^{-i\tau H_j/2}\Big)\Big(\prod_{j=m}^1 e^{-i\tau H_j/2}\Big)$,
以及 $p$ 阶 \textbf{Suzuki 组合公式}.

\begin{Thm}[一阶 Trotter 的交换子型误差界]
    设 $H = H_1 + H_2$, $\tau = t/L$. 由 Duhamel 原理(Model1)与交换子估计
    $\norm{[e^{-isH_1}, H_2]} \le s\norm{[H_1,H_2]}$:
    \begin{equation}
        \norm{e^{-i\tau H} - e^{-i\tau H_1}e^{-i\tau H_2}}
        \le \frac{\tau^2}{2}\norm{[H_1,H_2]}.
    \end{equation}
    对 $m$ 项情形 $H = \sum_j H_j$, 全局误差为
    \begin{equation}
        \norm{e^{-itH} - S_1(t/L)^L} \le \frac{t^2}{2L}\sum_{j<k}\norm{[H_j, H_k]},
    \end{equation}
    故达到精度 $\varepsilon$ 所需步数
    \begin{equation}
        L = O\!\Big(\frac{t^2\,\sum_{j<k}\norm{[H_j,H_k]}}{\varepsilon}\Big)
        \le O\!\Big(\frac{(t\Lambda)^2}{\varepsilon}\Big),
        \qquad \Lambda := \sum_j \norm{H_j}.
    \end{equation}
    交换子界优于朴素的范数界 $\norm{[H_j,H_k]} \le 2\norm{H_j}\norm{H_k}$:
    不相交的项(交换子为零)不贡献误差.
\end{Thm}

\begin{Thm}[高阶 Suzuki 公式]
    二阶(Strang 分裂)局部误差 $O(\tau^3)$, $p$ 阶公式局部误差 $O(\tau^{p+1})$,
    达到精度 $\varepsilon$ 所需步数
    \begin{equation}
        L = O\!\Big(\frac{(t\Lambda)^{1+1/p}}{\varepsilon^{1/p}}\Big),
    \end{equation}
    每步调用 $O(m\cdot 5^{\lceil p/2\rceil-1})$ 次局部演化($m$ 为项数; Suzuki 递推
    使指数个数随阶数 $p = 2k$ 按 $5^{k-1}$ 增长). 权衡: 高阶改进了对 $t,\varepsilon$ 的依赖,
    但每步常数变大, 实践中 $p = 2, 4, 6$ 往往最优.
\end{Thm}

\begin{Note}[局部演化的实现]
    每个因子 $e^{-i\tau H_j}$ 仍需分解为基本门. 若 $H_j = c\,P$(单个 Pauli 串),
    用 CNOT 链计算奇偶性后施加单个 $R_z(2c\tau)$ 再还原(共 $2(k-1)$ 个 CNOT 与 1 个 $R_z$,
    见 Module2 的 $ZZ$ 电路); 若 $H_j$ 是若干对易项之和, 可直接分解为各对易项指数的乘积
    (\textbf{快速前推}). 对 PDE 离散矩阵, $H_j$ 通常对应移位算子或对角算子,
    其指数可由移位电路/对角相位门实现.
\end{Note}

\begin{Eg}[横向场 Ising 模型: 交换子界优于范数界]
    对 $H = -J\sum_{i=1}^{n-1}Z_iZ_{i+1} - g\sum_{i=1}^n X_i = H_1 + H_2$:
    $\norm{H_1},\norm{H_2} = O(n)$ 给出范数界 $L = O(n^2t^2/\varepsilon)$;
    而 $[Z_iZ_{i+1},X_k] \ne 0$ 仅当 $k = i$ 或 $k = i+1$, 故
    $\norm{[H_1,H_2]} = O(n)$, 交换子界给出更紧的 $L = O(nt^2/\varepsilon)$
    (细节见 Module2).
\end{Eg}

\begin{Note}[随机化乘积公式(randomized product formulas)]
    Childs--Ostrander--Su(2019)证明: 对 $p$ 阶公式做随机化(每步随机选取分解次序/截断),
    期望代价可改进为
    \begin{equation}
        \widetilde O\!\Big(\frac{t^{1+1/(2p)}}{\varepsilon^{1/(2p)}}\Big)
    \end{equation}
    次局部演化(相对确定性公式 $O(t^{1+1/p}\varepsilon^{-1/p})$ 是指数的\textbf{平方根}改进).
    代价是成功概率概率性保证(误差为期望意义), 电路结构仍为乘积公式, 适合 NISQ.
\end{Note}

\begin{Note}[Trotter 的定位]
    优点: 电路结构就是"轮流演化各局部项", 辅助比特为 $O(1)$,
    适合\textbf{含噪中等规模量子(Noisy Intermediate-Scale Quantum, NISQ)};
    缺点: 误差对 $t$ 是多项式(非指数)依赖, 长时演化代价高;
    且要求每个 $e^{-i\tau H_j}$ 本身容易实现.
\end{Note}

\subsection{Taylor 级数法与 LCU}

把指数函数展开为\textbf{Taylor 级数}并截断, 再以\textbf{线性组合酉(LCU)}(Module3)合成:

\begin{equation}
    e^{-iHt} = \sum_{k=0}^{\infty} \frac{(-iHt)^k}{k!}
    \;\approx\; \sum_{k=0}^{K} \frac{(-it)^k}{k!} H^k,
\end{equation}

其中每项 $H^k$ 可用 $k$ 次 $U_H$ 的受控版本实现. 若\textbf{直接对整段时间} $t$ 做单段截断, 有两重困难:
首先, 当 $k \lesssim \tau := \Lambda t$ 时 $\tau^k/k!$ 仍随 $k$ \textbf{递增}(需越过 $k \sim e\tau$ 的峰值才开始衰减),
故截断阶需 $K = O\!\big(\tau + \frac{\log(1/\varepsilon)}{\log\log(1/\varepsilon)}\big)$ 量级
($\tau$ 项来自"越过峰值", 精度项来自峰后的超指数衰减);
其次, 也是更本质的, LCU 的归一化因子 $\sum_{k\le K}\tau^k/k! \approx e^{\tau}$ \textbf{指数}大,
直接后选择成功概率 $\sim e^{-2\tau}$ 指数小, 需要指数多次振幅放大, 不可行.

\begin{Note}[LCU 引理(Module3 回顾)]
    设 $M = \sum_j w_j V_j$($w_j \ge 0$, $V_j$ 酉), $s = \sum_j w_j$.
    构造 \textbf{PREPARE} $V\ket{0^a} = \sum_j\sqrt{w_j/s}\ket j$ 与
    \textbf{SELECT} $\mathrm{SEL} = \sum_j\ket j\bra j\otimes V_j$, 则
    \begin{equation}
        (V^\dagger\otimes I)\,\mathrm{SEL}\,(V\otimes I)\ket{0^a}\ket\psi
        = \ket{0^a}\otimes\frac{M\ket\psi}{s} + \ket{0^a}^{\perp}\otimes(\cdots),
    \end{equation}
    测量辅助得到 $\ket{0^a}$ 的概率为 $\norm{M\ket\psi}^2/s^2$,
    成功时系统态恰为 $M\ket\psi/\norm{M\ket\psi}$.
\end{Note}

BCCKS 的做法是把时间分为
$r = \lceil\tau\rceil$ 段(每段长度 $t/r$、$\Lambda t/r \le 1$), 每段截断阶数
$K = O\!\big(\frac{\log(\tau/\varepsilon)}{\log\log(\tau/\varepsilon)}\big)$, 使每段误差 $\le\varepsilon/r$、
归一化因子 $O(1)$, 再用非感知振幅放大把成功概率提升到常数; 总查询复杂度为

\begin{equation}
    O\!\Big(\tau\, \frac{\log(\tau/\varepsilon)}{\log\log(\tau/\varepsilon)}\Big),
\end{equation}

这是 Berry--Childs--Cleve--Kothari--Somma 的截断 Taylor 级数方法.
其优点是精度指数依赖(乘性因子 $\log(1/\varepsilon)$), 对时间 $t$ \textbf{线性},
除 $\log\log$ 因子外已达无快速前推下界 $\Omega(t)$;
缺点是 \textbf{选择预言机(select oracle)}需要对 $H^k$ 的幂做受控, 电路较深,
且成功概率需要振幅放大补偿.

\subsection{Qubitization 与 QSVT 方法}

回顾 Module3: 对 $H$ 的 $(\alpha, m)$-block-encoding $U_H$
($\alpha \ge \norm{H}$, $(\bra{0^m}\otimes I)U_H(\ket{0^m}\otimes I) = H/\alpha$),
qubitization 迭代把每个本征方向化为旋转: 定义成功子空间反射
$Z = 2\ket{0^m}\bra{0^m}\otimes I - I$ 与\textbf{qubitization 迭代算子}
$O = U_H Z$(厄米情形), 则
\begin{equation}
    (\bra{0^m}\otimes I)\,O^k\,(\ket{0^m}\otimes I) = T_k(H/\alpha),
\end{equation}
其中 $T_k$ 为第一类 Chebyshev 多项式($T_k(\cos\theta) = \cos(k\theta)$).
于是矩阵函数的实现归结为把目标函数 $e^{-iHt}$ 写成 Chebyshev 级数.

\begin{Note}[Jacobi--Anger 展开与复杂度]
    令归一化谱变量 $x = \lambda/\alpha \in [-1,1]$, 由 Jacobi--Anger 展开($J_k$ 为第一类
    Bessel 函数, $\tau = \alpha t$):
    \begin{equation}
        e^{-i\tau x} = J_0(\tau) + 2\sum_{k=1}^{\infty}(-i)^k J_k(\tau)\,T_k(x)
        \;\Longrightarrow\;
        e^{-iHt} = J_0(\alpha t)I + 2\sum_{k=1}^{\infty}(-i)^k J_k(\alpha t)\,T_k(H/\alpha).
    \end{equation}
    Bessel 系数在 $k \gtrsim \alpha t$ 后超指数衰减($J_k(\tau) \approx (\tau/2)^k/k!$),
    截断到 $d$ 阶, 用 qubitization 逐项实现 $T_k(H/\alpha)$ 并经 QSP 相位因子做
    相干组合, 得到
    \begin{equation}
        \text{电路深度/查询数} = O\!\Big(\alpha t +
        \frac{\log(1/\varepsilon)}{\log\log(1/\varepsilon)}\Big),
    \end{equation}
    其中 $\alpha t$ 项来自"至少需要 $\sim\alpha t$ 个 Chebyshev 项",
    $\log(1/\varepsilon)/\log\log(1/\varepsilon)$ 项来自 Bessel 系数的超指数衰减.
    该复杂度匹配下界 $\Omega(\alpha t + \log(1/\varepsilon)/\log\log(1/\varepsilon))$
    (§3.1 无快速前推 + 精度下界), 是目前的\textbf{最优方法}: 时间与精度为\textbf{加性}关系.
\end{Note}

\begin{Note}[与 Trotter 的本质区别]
    Trotter 属于\textbf{时间域算子分裂 + 时间步进}: 误差来自非对易项, 对 $t$ 为
    超线性依赖($t^{1+1/p}$); Qubitization/QSP 属于\textbf{谱域多项式变换}:
    先经 block-encoding 把特征值映射成旋转角, 再在频谱上用 Chebyshev 多项式逼近,
    对 $t$ 线性、对 $\log(1/\varepsilon)$ 加性 —— 这是两种范式的分水岭
    (细节与 QSP 表示定理见 Module3).
\end{Note}

\subsection{含时哈密顿量(补充)}

对\textbf{时间依赖}的 $H(t)$, 解是\textbf{时间有序指数(time-ordered exponential)}
$U(t) = \mathcal{T}\exp\big(-i\int_0^t H(s)\,ds\big)$, 不能写成 $e^{-iHt}$.
\begin{Note}[相互作用绘景(interaction picture)]
    设 $H(t) = A + B(t)$, 其中 $A$ 时间无关且易于模拟(可快速前推或廉价模拟),
    $B(t)$ 为含时部分. Low--Wiebe(2018)证明: 在相互作用绘景中模拟 $e^{-iHt}$ 的代价
    与 $\int_0^t \norm{B(s)}\,ds$ 成比例, 而非 $t(\norm{A} + \max_s\norm{B(s)})$:
    当 $B$ 是 $A$ 的小扰动(如薛定谔化后 $H_{\mathrm{eff}}(\xi) = \xi H_1 + H_2$
    中 $\norm{H_2}$ 远小于 $\norm{H_1}$ 的情形)时显著更优.
\end{Note}
\begin{Note}[与薛定谔化管线的衔接]
    薛定谔化把时间依赖系统化为 $i\partial_t v = H_{\mathrm{eff}}(t)v$ 后,
    处理含时 $H_{\mathrm{eff}}(t)$ 有两条路线: (1) 本模块 §4.6 的
    \textbf{autonomous technique}(引入 $s$ 变量把非自治问题化为自治问题);
    (2) 相互作用绘景/含时乘积公式. 二者对 PDE 问题各有适用性:
    前者需要额外维度, 后者要求 $H_{\mathrm{eff}}$ 可分裂为"易模拟主项 + 小扰动".
\end{Note}

\begin{Note}[补充: 热半群(实指数)的 QSVT —— 与振荡传播子的对比]
    上面的综述针对酉演化 $e^{-iHt}$; 对抛物型问题需要的是\textbf{实指数}
    $E_T = e^{-TL}$($L = L^\dagger \succeq 0$, 如热方程离散后的 $L_h$). 设 $L/\alpha$ 有
    Hermitian block-encoding、$\mathrm{spec}(L/\alpha)\subseteq[0,1]$, 记 $\tau := T\alpha$,
    则目标是标量函数 $x \mapsto e^{-\tau x}$, $x \in [0,1]$(Xiantao Li 讲义 §4.3).
    QSVT 实现它有一个\textbf{关键细节}: QSP/QSVT 的容许多项式必须在\textbf{整个} $[-1,1]$ 上有界,
    而 $e^{-\tau x}$ 在 $[-1,0]$ 上增长到 $e^\tau$, 不能直接逼近. 标准补救是\textbf{仿射移位}:
    令 $B := I - L/\alpha$($\mathrm{spec}(B)\subseteq[0,1]$; 仿射变换 $x\mapsto 1-x$ 对
    block-encoding 是免费的, 不改变渐近查询数), 则
    $e^{-TL} = e^{-\tau(I-B)} = f(B)$ 对应的标量函数
    \begin{equation}
        f(x) = e^{-\tau(1-x)}, \qquad e^{-2\tau} \le f(x) \le 1 \quad (x \in [-1,1]),
    \end{equation}
    与 QSVT 的全局有界性相容. 相应多项式次数与查询数为
    \begin{equation}
        m = O\Big(\sqrt{\max\{\tau,\ \log(1/\varepsilon)\}\cdot\log(1/\varepsilon)}\Big),
        \qquad
        Q = \widetilde O\Big(\sqrt{T\alpha\,\log(1/\varepsilon)} + \log(1/\varepsilon)\Big),
    \end{equation}
    对二阶差分 $\alpha = \Theta(h^{-2})$ 得 $Q = \widetilde O\big(\frac{\sqrt T}{h}\sqrt{\log(1/\varepsilon)}\big)$.

    \textbf{平方根标度的来源}: Chebyshev 多项式 $T_m$ 的极值在区间端点\textbf{二次聚集}, 宽度
    $\Theta(1/m^2)$ 的端点结构只需度 $m \sim \sqrt{1/\text{宽度}}$; 热滤波器 $e^{-\tau x}$ 是
    $x=0$ 处宽度 $\sim 1/\tau$ 的\textbf{边界层}(从 1 掉到 $\varepsilon$ 的谱窗口长
    $\log(1/\varepsilon)/\tau$, 之后平坦), 故 $m \sim \sqrt{\tau}$. 而振荡滤波器 $e^{-itx}$
    在区间内处处以尺度 $1/t$ 振荡, 多项式必须整体分辨, 度数与 $t$ 成正比 —— 这是抛物问题与
    哈密顿量模拟在时间-精度标度上的本质区别(哈密顿量模拟下界 $\Omega(t)$ 不适用于热半群).

    \textbf{实现算符 vs 制备归一化解}: 上述 $Q$ 是实现热算符 $E_T$ 的查询数. 若目标是制备
    \textbf{归一化}终态 $E_T\ket{y_0}/\norm{E_T\ket{y_0}}$, 后选择成功概率
    $p_T = \norm{E_T\ket{y_0}}^2$, 需额外因子 $g_T = p_T^{-1/2} = \norm{y_h(0)}/\norm{y_h(T)}$;
    Dirichlet 边界下最小本征值 $\lambda_{1,h} = \Theta(1)$, 长时间演化可使 $g_T$ 随 $T$ 指数增长 ——
    实现热算符与产生归一化热态是两回事(后者与 4.4 节恢复概率的分析类似).
\end{Note}

\begin{center}
\begin{tabular}{@{}lclcl@{}}

    \midrule
    Trotter/Suzuki & $O\big((t\Lambda)^{1+1/p}\varepsilon^{-1/p}\big)$ & 多项式 & $O(1)$ & NISQ, 短时 \\

    Qubitization/QSVT & $O\!\big(\alpha t + \frac{\log(1/\varepsilon)}{\log\log(1/\varepsilon)}\big)$ & 指数 & $O(m+1)$ & 容错, 最优 \\

\end{tabular}

\begin{Note}[查询下界视角的排序]
    四种方法对 $t$ 的依赖依次为 $t^{1+1/p} \to t^{1+1/(2p)} \to t \to t$;
    对 $1/\varepsilon$ 的依赖依次为 $\varepsilon^{-1/p} \to \varepsilon^{-1/(2p)}
    \to \log \to \log\log$. Qubitization/QSVT 同时达到 $t$ 与 $\varepsilon$ 的最优
    渐近依赖, Taylor+LCU 只差 $\log\log$ 因子.
\end{Note}

\begin{Note}[薛定谔化管线中的选择]
    薛定谔化把 PDE 化为 $i\partial_t v = H_{\mathrm{eff}} v$ 后, $H_{\mathrm{eff}}$
    通常可写成"移位算子(shift operator)与对角算子(diagonal operator)之和"的形式, 自然获得 block-encoding,
    因此 QSVT 路线(或对其结构定制的高阶 Trotter)是首选;
    对含噪中间尺度设备, Trotter 仍是务实的备选.
    特别地, 当 $H_{\mathrm{eff}}$ 的对角部分与移位部分交换时, 乘积公式退化为精确快速前推
    (§3.2 对易情形), 无需任何分裂误差.
\end{Note}

\section{薛定谔化方法}

\subsection{动机: PDE 的量子模拟}

一般的线性\textbf{发展方程(evolution equation)}(\textbf{热方程(heat equation)}、\textbf{对流扩散方程(advection--diffusion equation)}、\textbf{Fokker--Planck 方程(Fokker--Planck equation)}、\textbf{Black--Scholes 方程(Black--Scholes equation)}等)
在\textbf{空间离散化(spatial discretization)}后具有形式

\begin{equation}
    \frac{du}{dt} = L u, \qquad \text{或更一般地} \quad
    \frac{du}{dt} = A(t) u + f(t),
\end{equation}

其中 $L$ 或 $A$ 一般\textbf{不是厄米矩阵}(热方程: $L$ 厄米但演化 $e^{-tL}$\textbf{非酉(non-unitary)};
\textbf{对流方程(advection equation)}: $A$ 甚至\textbf{反厄米(anti-Hermitian)}). 量子模拟的困难正在于此: 标准的哈密顿量模拟
只处理 $i\partial_t u = Hu$ 型($H$ 厄米、演化酉)的方程.

\begin{Note}[两条历史路线]
    \begin{enumerate}
        \item \textbf{线性化 + 线性方程组}: 对 $u_t = Lu$ 做时间离散
        (如\textbf{隐式欧拉(implicit Euler)} $(I + \Delta t L)u^{n+1} = u^n$), 每步调用 HHL 求解线性方程组,
        共 $M = T/\Delta t$ 步 —— 需要 $M$ 次线性方程组求解, 每次都要
        block-encoding 与误差控制;
        \item \textbf{薛定谔化(Schrödingerisation)}: 把 $u_t = Lu$ \textbf{整体}转化为
        一个薛定谔方程, 只做\textbf{一次}哈密顿量模拟, 无需时间离散.
    \end{enumerate}
    本节的薛定谔化方法由 Jin--Liu--Yu 于 2023--2024 年提出, 是第二条路线的现代实现.
\end{Note}

\subsection{Warped phase transformation: 半正定情形}

\begin{Thm}[薛定谔化: 热方程型] \label{thm:schrodingerization}
    设 $u_t = -Lu$, 其中 $L = L^\dagger \succeq 0$($L$ 半正定厄米, 例如
    $L = -\Delta$ 加边界条件). 引入\textbf{辅助变量(auxiliary variable)} $p \ge 0$ 与\textbf{warped phase transformation}
    \begin{equation}
        v(t, p) = e^{-p}\, u(t),
    \end{equation}
    则 $v$ 满足
    \begin{equation}
        \partial_p v = -v, \qquad
        \partial_t v = -Lv = L\,\partial_p v.
    \end{equation}
    对 $p$ 做\textbf{Fourier 变换(Fourier transform)}. 本文统一采用约定
    $\hat f(\xi) := \int_{\mathbb R} f(p)\,e^{-i\xi p}\,dp$, 于是
    $\widehat{\partial_p f}(\xi) = i\xi\,\hat f(\xi)$(分部积分), 即 $\partial_p \leftrightarrow i\xi$:
    \begin{equation}
        i\, \partial_t \hat{v}(t, \xi) = -\xi L\, \hat{v}(t, \xi).
    \end{equation}
    这是哈密顿量为 $H(\xi) = -\xi L$ 的薛定谔方程; 对每个 $\xi \in \mathbb R$,
    $H(\xi)$ 是厄米算子($\xi$ 实数、$L$ 厄米), 演化 $e^{-iH(\xi)t}$ 是酉的. 原解由
    \begin{equation}
        u(t) = e^{p} v(t, p) \quad \text{对任意 } p \ge 0, \qquad
        \text{或} \quad
        u(t) = \int_0^\infty v(t, p)\, dp
    \end{equation}
    恢复($\int_0^\infty e^{-p}dp = 1$).
\end{Thm}

\begin{Proof}
    由 $v = e^{-p}u$ 直接求导: $\partial_p v = -e^{-p}u = -v$,
    以及 $\partial_t v = e^{-p}(-Lu) = -Lv$. 另一方面
    $L\partial_p v = L(-v) = -Lv$, 故 $\partial_t v = L\partial_p v$.
    Fourier 变换后 $\partial_p \to i\xi$, 得
    $\partial_t \hat v = L \cdot (i\xi)\hat v = i\xi L \hat v$,
    两边乘 $i$ 即 $i\partial_t\hat v = -\xi L\hat v$.
    $H(\xi) = -\xi L$ 的厄米性来自 $\xi \in \mathbb{R}$ 与 $L$ 厄米.
\end{Proof}

\begin{Note}[为什么叫"warped phase"]
    $v = e^{-p}u$ 相当于在振幅上附加了指数衰减权重 $e^{-p}$;
    $p$ 方向的"相位扭曲"在 Fourier 变换后把非酉算子 $L$ 变成了厄米算子
    $-\xi L$, 代价是增加一个维度(辅助变量 $\xi$).
    直观上: \textbf{耗散演化(dissipative evolution)}被"提升"到高维空间后变为保范演化, 通过额外的维度
    记录耗散的历史.
\end{Note}

\subsection{一般线性系统}

对一般的(非厄米、\textbf{非自治(non-autonomous)}、带\textbf{源项(source term)})系统, 薛定谔化按如下方式推广:

\begin{Thm}[一般情形的薛定谔化]
    设 $u_t = A(t)u + f(t)$, 把 $A$ 分解为厄米部分与反厄米部分
    \begin{equation}
        A = H_1 - iH_2, \qquad
        H_1 = \frac{A + A^\dagger}{2}, \qquad
        H_2 = \frac{i(A - A^\dagger)}{2},
    \end{equation}
    即 $H_1$, $H_2$ 均厄米. 这里 $H_1$ 对应增长/耗散部分, $H_2$ 对应相位旋转部分;
    \textbf{不要求} $-H_1 \succeq 0$. $H_1$ 的正特征值只改变恢复区间
    (见下一小节), 不会破坏薛定谔化本身.
    做 warped phase transformation $v = e^{-p}u$($\partial_p v = -v$, 即 $v = -\partial_p v$).
    关键观察: $-iH_2$ 部分本身是\textbf{酉演化的生成元}, 无需提升到 $p$ 方向;
    只有 $H_1$($A$ 的非酉部分)需要与 $p$ 耦合. 于是把 $H_1$ 项代入 $p$ 导数、$H_2$ 项保留:
    \begin{equation}
        \partial_t v = Av = (H_1 - iH_2)v = -H_1\,\partial_p v - iH_2 v.
    \end{equation}
    采用与定理 \ref{thm:schrodingerization} 相同的 Fourier 约定
    ($\widehat{\partial_p f} = i\xi\hat f$), 对 $p$ 做 Fourier 变换:
    \begin{equation}
        \partial_t \hat v = -H_1(i\xi)\hat v - iH_2\hat v = -i\big(\xi H_1 + H_2\big)\hat v,
    \end{equation}
    两边乘 $i$ 得
    \begin{equation}
        i\, \partial_t \hat v(t, \xi) = \big(\xi H_1 + H_2\big)\, \hat v(t, \xi),
        \label{eq:schr-general}
    \end{equation}
    即每个频率分量 $\xi$ 对应一个薛定谔方程, 哈密顿量
    \begin{equation}
        H_{\mathrm{eff}}(\xi) = \xi H_1 + H_2
    \end{equation}
    对每个 $\xi \in \mathbb R$ 厄米($\xi$ 实数、$H_1,H_2$ 厄米).
    若采用相反的 Fourier 约定或 $A = H_1 + iH_2$ 的分解, 只改变 $\xi \mapsto -\xi$ 与
    $H_2$ 的整体符号, 不改变"每个频率分量对应一个厄米哈密顿量"这一核心事实.
\end{Thm}

\begin{Note}[为什么"只有 $H_1$ 与 $p$ 耦合"是合理的]
    对任意解 $v = e^{-p}u$ 而言, $\partial_p v = -v$ 是恒等式, 因此把 $H_1$ 项写成
    $-H_1\partial_p v$ 不改变提升方程的解. 若把 $H_2$ 项也代入($-iH_2v = iH_2\partial_p v$),
    则 Fourier 后得到 $H_{\mathrm{eff}} = \xi(H_1 - iH_2) = \xi A$, 一般不厄米
    (这就是"朴素提升"失败的根源). 正确的选择是把 $A$ 的\textbf{反厄米部分
    (酉生成元)保留为 Hamiltonian 的组成部分}、只把厄米部分(耗散/增长)与 $p$ 耦合,
    从而得到厄米的 $H_{\mathrm{eff}}(\xi) = \xi H_1 + H_2$.
    验证: $e^{-p}u(t)$ 仍满足提升方程
    $\partial_t v = -H_1\partial_p v - iH_2 v$(代入 $\partial_p v = -v$ 两边相等).
\end{Note}

\begin{Note}[三种情形的处理]
    \begin{enumerate}
        \item \textbf{非厄米部分}: $A$ 的反厄米部分被吸收进 $H_2$;
        若需要更一般的处理, 也可以用\textbf{厄米膨胀}
        $H_A = \begin{pmatrix} 0 & A \\ A^\dagger & 0 \end{pmatrix}$
        把非厄米矩阵嵌入厄米矩阵;
        \item \textbf{非齐次项(non-homogeneous term) $f(t)$}: 用 \textbf{Duhamel 原理(Duhamel's principle)}
        (Module1)把源项化为对演化算子的积分, 再以 LCU 合成;
        \item \textbf{非自治系统 $A(t)$}: 把时间 $t$ 视为额外维度
        (引入 $t$ 的副本), 将非自治问题化为高一个维度的\textbf{自治(autonomous)}问题.
    \end{enumerate}
\end{Note}

\subsection{恢复定理与恢复区间}

\begin{Thm}[恢复定理]
    设 $u_t = Au$ 的薛定谔化提升解为 $w(t,p)$, 即 $w$ 满足
    $\partial_t w = -H_1\partial_p w - iH_2 w$ 且初值
    $w(0,p) = \psi(p)\,u_0$, 其中辅助函数 $\psi(p) = e^{-p}$ 在区间 $(p_*, p^*)$ 上成立.
    令
    \begin{equation}
        \lambda_+ := \max\{\lambda_{\max}(H_1),\, 0\}, \qquad
        \lambda_- := \max\{-\lambda_{\min}(H_1),\, 0\},
    \end{equation}
    分别表示 $H_1$ 的最大正特征值与最大负特征值的绝对值. 则对任意
    \begin{equation}
        p \in I_T := \big(p_* + \lambda_+ T,\ p^* - \lambda_- T\big)
    \end{equation}
    都有 $u(T) = e^{p}\, w(T,p)$.
\end{Thm}

\begin{Proof}[三步证明]
    \textbf{第一步: 参考解.} 记 $\tilde w(t,p) := e^{-p}u(t)$. 上一节已验证它满足
    同一个提升方程 $\partial_t\tilde w = -H_1\partial_p\tilde w - iH_2\tilde w$
    (代入 $\partial_p\tilde w = -\tilde w$ 两边相等). 于是误差
    $e(t,p) := w(t,p) - \tilde w(t,p)$ 满足同型方程
    \begin{equation}
        \partial_t e = -H_1\,\partial_p e - iH_2 e,
    \end{equation}
    且由 $\psi = e^{-p}$ 于 $(p_*,p^*)$ 得
    $e(0,p) = w(0,p) - \tilde w(0,p) = \psi(p)u_0 - e^{-p}u_0 = 0$ 于 $p \in (p_*,p^*)$.
    问题归结为: \emph{一个在初始 $p$-区间上为零的解, 时刻 $T$ 在哪个区间上仍为零?}

    \textbf{第二步: 化为纯传输方程(有限传播速度).}
    $H_2$ 项 $-iH_2e$ 是 $p$-\textbf{局部}的(对 $e$ 的逐点作用, 不移动 $p$-支撑),
    但会混合不同 $x$-分量, 不能直接按分量讨论. 用酉变换 $e^{iH_2 t}$ 去掉该项:
    令 $z(t,p) := e^{iH_2 t} e(t,p)$, 则 $z(0,\cdot) = e(0,\cdot)$(支撑相同)且
    \begin{equation}
        \partial_t z = -e^{iH_2 t}H_1 e^{-iH_2 t}\,\partial_p z
        =: -A(t)\,\partial_p z,
    \end{equation}
    其中 $A(t) := e^{iH_2 t}H_1 e^{-iH_2 t}$ 与 $H_1$ \textbf{酉等价}, 故谱相同:
    $\mathrm{spec}(A(t)) = \mathrm{spec}(H_1) \subseteq [-\lambda_-, \lambda_+]$.
    这是(含时系数)传输方程: 每个 $x$-分量的 $p$-信息以 $A(t)$ 的特征值为传播速度,
    故所有 $p$-信息的传播速度都落在 $[-\lambda_-, \lambda_+]$ 内, $z$(从而 $e$)的
    $p$-支撑至多按速度 $\lambda_+$(向右)与 $\lambda_-$(向左)扩张.

    \textbf{第三步: 恢复区间.} 初始零区域 $(p_*, p^*)$ 两侧分别以速度
    $\lambda_+$、$\lambda_-$ 收缩, 时刻 $T$ 后仍为零的区域为
    \[
        I_T = (p_* + \lambda_+ T,\ p^* - \lambda_- T).
    \]
    故 $e(T,p) = 0$ 于 $p \in I_T$, 即 $w(T,p) = e^{-p}u(T)$,
    得 $u(T) = e^{p}w(T,p)$.
\end{Proof}

\begin{Note}[恢复区间的可行性]
    恢复区间 $I_T$ 非空要求 $p^* - p_* > (\lambda_+ + \lambda_-)T$:
    初值 $\psi$ 的"精确区间" $(p_*,p^*)$ 必须足够宽, 以容纳 $[0,T]$ 内的信息传播.
    对 $\psi(p) = e^{-p}$($p_* = 0$, $p^* = \infty$)与 $H_1 \preceq 0$
    (热方程情形, $\lambda_+ = 0$)有 $I_T = (0,\infty)$, 任意 $p > 0$ 都可恢复;
    若 $H_1$ 有正特征值, 则必须取 $p > \lambda_+ T$. 对有限的 $p^*$,
    需事先保证 $p^* > \lambda_- T$(如取 $\psi$ 在截断区间内衰减).
    这也解释了为什么 $H_1$ 的正特征值不破坏方法、只抬高恢复点 $p$ 的下界.
\end{Note}

\begin{Note}[特例与积分恢复]
    取 $\psi(p) = e^{-|p|}$ 时初始精确区间为 $(0, \infty)$; 若 $H_1 \preceq 0$
    (热方程情形), 则 $\lambda_+ = 0$, $I_T = (0,\infty)$, 任意 $p > 0$ 都可恢复.
    若 $H_1$ 有正特征值, 则必须取 $p > \lambda_+ T$.
    更稳定的恢复方式是\textbf{积分恢复}: 当 $w(T,q) = e^{-q}u(T)$ 对 $q \ge p$
    成立时,
    \begin{equation}
        u(T) = e^{p} \int_p^\infty w(T,q)\,dq.
    \end{equation}
    离散后测量 $p$-寄存器落在恢复指标集 $I^\diamond$ 中的成功概率近似为
    \begin{equation}
        P_{\mathrm{suc}} \approx \frac{C_{\mathrm{rec}}^2\, \norm{u(T)}^2}
        {C_\psi^2\, \norm{u_0}^2},
    \end{equation}
    其中 $C_\psi^2 = \sum_k |\psi(p_k)|^2$, $C_{\mathrm{rec}}^2 = \sum_{k \in I^\diamond} e^{-2p_k}$.
    振幅放大因子为 $O(\Gamma_\psi\, q)$, 其中
    $\Gamma_\psi := C_\psi/C_{\mathrm{rec}}$, $q := \norm{u_0}/\norm{u(T)}$
    是原始非酉演化的衰减因子.
\end{Note}

\subsection{薛定谔化的五步流程}

\begin{Def}[量子模拟 PDE 的标准流程]
    \begin{enumerate}
        \item \textbf{薛定谔化}: 用 warped phase transformation 把 $u_t = Lu$
        化为 $i\partial_t v = H_{\mathrm{eff}} v$(本模块核心);
        \item \textbf{空间离散化}: 对 $x$ 与辅助变量 $\xi$ 做\textbf{有限差分(finite difference)}/\textbf{谱离散(spectral discretization)}
        (注意 $\xi$ 的截断区间与网格由精度要求决定);
        \item \textbf{Block-encoding}: 把离散后的 $H_{\mathrm{eff}}$ 表示成
        移位算子与对角算子之和, 构造 $(\alpha, m)$-block-encoding;
        \item \textbf{哈密顿量模拟}: 用 Trotter 或 qubitization/QSVT 制备
        $\ket{\Psi(T)} = e^{-iH_{\mathrm{eff}}T}\ket{\Psi(0)}$;
        \item \textbf{测量读出}: 对 $p$(或 $\xi$)方向做\textbf{逆 Fourier 变换(inverse Fourier transform)},
        通过期望值测量提取 $u(x, T)$(需要 $O(1/\varepsilon^2)$ 次重复).
    \end{enumerate}
    其中步骤 3、4 分别对应 Module3 与本节前面的综述.
\end{Def}

\subsection{例子: 热方程}

\begin{Eg}[一维热方程]
    考虑 $u_t = u_{xx}$, $x \in (0,1)$, \textbf{Dirichlet 边界(Dirichlet boundary condition)} $u(0,t) = u(1,t) = 0$.
    二阶\textbf{中心差分(central difference)}离散化($h = 1/(N+1)$)给出 $u_t = -L_h u$, 其中
    \begin{equation}
        L_h = \frac{1}{h^2}
        \begin{pmatrix}
            2 & -1 & & \\ -1 & 2 & \ddots & \\ & \ddots & \ddots & -1 \\ & & -1 & 2
        \end{pmatrix},
    \end{equation}
    且 $L_h$ 的特征值为
    \begin{equation}
        \lambda_k(L_h) = \frac{4}{h^2}\sin^2\frac{k\pi}{2(N+1)}, \qquad k = 1, \dots, N.
    \end{equation}
    由定理 \ref{thm:schrodingerization}, 薛定谔化的哈密顿量为
    \begin{equation}
        H_{\mathrm{eff}}(\xi) = -\xi L_h,
    \end{equation}
    对每个 $\xi$ 厄米, 演化 $e^{-iH_{\mathrm{eff}}t}$ 可直接模拟.
    电路中 $\xi$ 的取值编码在一个辅助寄存器中, $H_{\mathrm{eff}}$ 的
    block-encoding 由 $L_h$ 的三对角结构(或移位算子分解)给出,
    随后施加 Trotter/QSVT 演化与逆 Fourier 变换读出 $u(x,T)$.
\end{Eg}

\begin{Note}[耗散与酉演化的关系]
    半离散热方程的解满足 $\norm{u(t)} \le \norm{u(0)}$(能量单调衰减),
    但 $e^{-tL_h}$ 本身非酉. 薛定谔化后 $e^{-iH_{\mathrm{eff}}t}$ 是酉的,
    耗散信息被"编码"进 $\xi$ 方向: 测量 $\xi$ 方向后, 概率集中在
    与原解一致的分量上 —— 这正是 $e^{-p}$ 权重的作用.
\end{Note}

\subsection{有限差分算子的 block-encoding: 移位算子 LCU}

五步流程的第 3 步要求把离散后的 $H_{\mathrm{eff}}$ 表示为"移位算子与对角算子之和"并构造
block-encoding. 这里以周期性网格上的标准差分算子为例, 给出\textbf{显式构造}
(Xiantao Li 讲义 §1.13; Camps--Lin--Van Beeumen--Yang 的带型循环矩阵构造).
设 $N = 2^n$, 定义\textbf{循环移位(circular shift)} $S\ket{j} = \ket{j+1 \pmod N}$,
$S^\dagger\ket{j} = \ket{j-1 \pmod N}$($S$ 的电路是模 $N$ 加一, $S^\dagger$ 是模 $N$ 减一;
不要与单比特相位门 $S$ 混淆). 一个差分算子即"若干个 $I$ 与 $\pm S^{\pm 1}$ 的线性组合",
故其 block-encoding 是 §3.3 LCU 的直接应用.

\textbf{向前差分.} 周期性向前差分 $(D_+ u)_j = (u_{j+1}-u_j)/h$ 满足 $D_+ = (S^\dagger - I)/h$,
即 $-D_+ = (I-S^\dagger)/h$. 取单比特选择寄存器,
$\mathrm{SELECT}_D = \ket{0}\bra{0}\otimes I + \ket{1}\bra{1}\otimes(-S^\dagger)$,
$\mathrm{PREP}_D = H$($\ket{0}\mapsto(\ket{0}+\ket{1})/\sqrt2$), 则
\begin{equation}
    U_D = (H\otimes I)\,\mathrm{SELECT}_D\,(H\otimes I),
    \qquad
    (\bra{0}\otimes I)U_D(\ket{0}\otimes I) = \frac{I-S^\dagger}{2} = \frac{h(-D_+)}{2},
\end{equation}
即 $U_D$ 是 $-D_+$ 的精确 $(2/h, 1, 0)$-block-encoding.

\textbf{一维三点 Laplacian.} 周期性正 Laplacian($(L_h u)_j = (2u_j-u_{j-1}-u_{j+1})/h^2$)满足
\begin{equation}
    L_h = \frac{2I - S - S^\dagger}{h^2},
\end{equation}
是带型循环矩阵($a_0 = 2/h^2$, $a_{\pm} = -1/h^2$). 取两比特选择寄存器,
\begin{equation}
    \mathrm{SELECT}_L = \ket{0}\bra{0}\otimes I + \ket{1}\bra{1}\otimes(-S)
        + \ket{2}\bra{2}\otimes(-S^\dagger) + \ket{3}\bra{3}\otimes I,
\end{equation}
并取 PREP 满足 $P_L\ket{0} = \frac{\sqrt2}{2}\ket{0} + \frac12\ket{1} + \frac12\ket{2}$
($\ket{3}$ 振幅为零 —— 两个单位项合并进一个槽), 则
\begin{equation}
    U_L = (P_L^\dagger\otimes I)\,\mathrm{SELECT}_L\,(P_L\otimes I),
    \qquad
    (\bra{0}\otimes I)U_L(\ket{0}\otimes I) = \frac{2I-S-S^\dagger}{4} = \frac{h^2 L_h}{4},
\end{equation}
即 $U_L$ 是 $L_h$ 的精确 $(4/h^2, 2, 0)$-block-encoding. 这正给出本模块热方程例子中
三对角 $L_h$ 的 block-encoding: 该例的 $L_h$ 是非周期修正后的形式(见下面的边界注).

\textbf{高维推广.} 二维五点 Laplacian(等距 $h$)为
\begin{equation}
    L_h^{(2D)} = \frac{4I - S_x - S_x^\dagger - S_y - S_y^\dagger}{h^2},
    \qquad S_x = S\otimes I,\ \ S_y = I\otimes S,
\end{equation}
即 5 个不同酉项(把 $4I$ 视为 4 个单位项后共 8 个槽位)、权重和 $8/h^2$, 仍是同一 LCU 结构.

\begin{Note}[边界条件]
    周期性移位假设了循环边界. 对 Dirichlet/Neumann 等非周期边界, 需把"跨边界"的移位项修正掉:
    在结构化稀疏路线中, 通过边界受控旋转或对带型循环电路做非周期修正, 把 wrap-around 项清零
    (Camps--Lin--Van Beeumen--Yang). 本模块热方程例子中的 $L_h$ 正是三对角非周期矩阵,
    即该修正后的形式.
\end{Note}

\subsection{例子: 对流方程与迎风格式}

\begin{Eg}[一维对流方程]
    考虑 $u_t + c\,u_x = 0$, $c > 0$. \textbf{迎风(upwind)差分}
    $du_j/dt = -\frac{c}{h}(u_j - u_{j-1})$ 给出
    \begin{equation}
        u_t = A u, \qquad A = -\frac{c}{h}\,(I - S_-),
    \end{equation}
    其中 $S_-$ 为向后移位算子. 迎风格式数值稳定但 $A$ 非厄米; 分解
    \begin{equation}
        A = \frac{A + A^\dagger}{2} + \frac{A - A^\dagger}{2} = S + K,
    \end{equation}
    $S = S^\dagger$ 厄米, $K = -K^\dagger$ 反厄米, 二者都是移位算子的线性组合,
    block-encoding 容易构造. 按 $A = H_1 - iH_2$ 取 $H_1 = S$,
    $H_2 = iK$, 由式 \eqref{eq:schr-general} 薛定谔化后的哈密顿量为
    \begin{equation}
        H_{\mathrm{eff}}(\xi) = \xi S + iK,
    \end{equation}
    对每个 $\xi$ 厄米($\xi$ 实数、$S$ 与 $iK$ 厄米), 可直接模拟.
\end{Eg}

\subsection{Smooth initialization 与最优精度依赖}

\begin{Note}[初值光滑性是精度的主要瓶颈]
    若直接取 $\psi(p) = e^{-|p|}$, 该函数在 $p = 0$ 处不可微. 在 $p$ 方向使用
    Fourier 谱离散时, 函数越光滑谱逼近越快; 对 $e^{-|p|}$ 一般需要
    $\Delta p = O(\varepsilon)$, 因而 $\mu_{\max} \sim 1/\Delta p = O(1/\varepsilon)$.
    这使总查询复杂度对 $\varepsilon$ 呈多项式依赖, 瓶颈不在 Hamiltonian simulation,
    而在辅助变量初值的低光滑性.
\end{Note}

\begin{Note}[Smooth initialization 的三个条件]
    设 $\psi(p) = \varphi(p)\,e^{-p}$, 其中 $\varphi$ 是 $0 \to 1$ 的平滑阶跃函数.
    为获得接近最优的精度依赖, 需要:
    \begin{enumerate}
        \item \textbf{截断兼容}: $\psi$ 在计算区间外快速衰减, 使无限区间可以被
        有限区间周期近似;
        \item \textbf{恢复兼容}: 在恢复区域 $p \ge p^*$ 上
        $|\psi(p) - e^{-p}| \le \varepsilon$;
        \item \textbf{导数增长控制}: 对 $r \simeq \log(1/\varepsilon)$,
        $\norm{\psi^{(r)}}_{L^2}^{1/r} \le C\, r$.
    \end{enumerate}
    三者的量化作用: 条件 ① 控制 $p$ 方向\textbf{截断区间}的长度
    ($p$ 网格数 $N_p$); 条件 ② 直接给出\textbf{恢复误差}
    $\norm{u(T) - e^{p^\diamond}w(T,p^\diamond)} = O(\varepsilon)$;
    条件 ③ 控制 $\psi$ 的\textbf{Fourier 谱}衰减率, 从而决定 $p$ 方向谱离散所需的最大频率
    $\mu_{\max}$: 对 $r$ 阶可微函数, $\hat\psi(\xi) = O(|\xi|^{-r})$,
    要截断误差 $\le\varepsilon$ 需 $\mu_{\max} = O(\varepsilon^{-1/r})$;
    若 $\|\psi^{(r)}\|$ 的界对 $r$ 只按 $C^r r^r$(\textbf{Gevrey-1/解析型})增长,
    则取 $r \sim \log(1/\varepsilon)$ 得 $\mu_{\max} = O(\log(1/\varepsilon))$ ——
    这是"光滑性以对数代价换取 $\mu_{\max}$ 从 $O(1/\varepsilon)$ 降到
    $O(\log(1/\varepsilon))$"的定量机制.
    仅 $\psi \in C^\infty$ 不够, 还必须控制高阶导数的增长速度;
    光滑 mollifier 通常只给出
    $\mu_{\max} = O((\log(1/\varepsilon))^{1/\beta})$($\beta < 1$).
\end{Note}

\begin{Note}[Error-function 初始化]
    取
    \begin{equation}
        \varphi(p) = \frac{\mathrm{erf}(a p) + 1}{2}, \qquad
        \psi(p) = \varphi(p)\,e^{-p},
    \end{equation}
    其中 $a \sim \sqrt{\log(1/\varepsilon)}$, $p^* \lesssim 1/2$.
    由 $\mathrm{erfc}(x) \lesssim e^{-x^2}$ 可得恢复误差 $O(\varepsilon)$;
    $\mathrm{erf}^{(k)}(ap) \sim a^k H_{k-1}(ap)\,e^{-a^2p^2}$
    给出解析型导数控制, 从而
    \begin{equation}
        \mu_{\max} = O\!\Big(\log\frac{1}{\varepsilon}\Big).
    \end{equation}
    若 $H_1$, $H_2$ 的 block-encoding 归一化尺度至多为 $\alpha_A$, 则一次
    Hamiltonian simulation 的矩阵查询数为
    \begin{equation}
        \widetilde O\!\Big(\Gamma_\psi\, q\, \alpha_A\, T\, \mu_{\max}\Big)
        = \widetilde O\!\Big(\Gamma_\psi\, q\, \alpha_A\, T\,
        \log\frac{1}{\varepsilon}\Big),
    \end{equation}
    这在下界 $\Omega(\alpha_A T)$ 与 $\Omega(\log(1/\varepsilon))$ 的意义下
    达到 matrix-query 最优.
\end{Note}

\subsection{带源项与时间依赖系统}

\begin{Note}[带源项: 增广齐次化]
    对 $u_t = A(t)u + b(t)$, 引入常向量 $r(t)$($\dot r = 0$)与矩阵 $B(t)$
    使 $B(t)r_0 = b(t)$, 令
    \begin{equation}
        u_f = \begin{pmatrix} u \\ r \end{pmatrix}, \qquad
        A_f(t) = \begin{pmatrix} A(t) & B(t) \\ 0 & 0 \end{pmatrix},
    \end{equation}
    则 $u_f$ 满足齐次系统 $\dot u_f = A_f u_f$, 原解是 $u_f$ 的第一块.
    常用归一化取 $\gamma_i = T\sup_{t\in[0,T]}|b_i(t)|$, $r_0 = \gamma$,
    $B(t) = \mathrm{diag}(b_i(t)/\gamma_i)$, 使 $\norm{B} \le O(1/T)$.
    时间无关源 $b(t) \equiv b$ 可更简单地取
    $u_f = (u, Tb)^T$, $A_f = \begin{pmatrix} A & I/T \\ 0 & 0 \end{pmatrix}$.
    源项使扩展后的 $H_1$ 不再保持半负定: 由 \textbf{Weyl 扰动定理(Weyl's perturbation theorem)},
    \[
        \lambda_{\max}(\tilde H_1) \le \lambda_+^{\max}(H_1^A) + \delta_B,
        \qquad \delta_B := \norm{B}/2,
    \]
    故恢复点可取 $p^\diamond = \big(\lambda_+^{\max}(H_1^A) + \delta_B\big)T$
    (时间无关源时 $\delta_B = 1/(2T)$, 即 $p^\diamond \le \lambda_+^{\max}(H_1^A)T + 1/2$).
\end{Note}

\begin{Note}[非自治系统: autonomous technique]
    薛定谔化后的时间依赖系统形如 $\dot w = -iH(t)w$, $H(t) = H(t)^\dagger$.
    由于 $[H(t_1), H(t_2)] \neq 0$, 形式解是\textbf{时间有序指数(time-ordered exponential)}
    $U_{t,s} = \mathcal{T}\exp(-i\int_s^t H(\tau)d\tau)$, 不能直接 fast-forward.
    引入额外变量 $s \in \mathbb{R}$ 并考虑自治 PDE
    \begin{equation}
        \partial_t v + \partial_s v = -iH(s)v, \qquad
        v(0,s) = G(s)w_0.
    \end{equation}
    沿特征线 $s - t = \text{const}$ 得
    \begin{equation}
        v(t,s) = G(s-t)\, U_{s,s-t}\, w_0,
    \end{equation}
    取 $G = \delta(s)$ 时 $w(t) = \int v(t,s)\,ds = U_{t,0}w_0$.
    用动量算子 $P_s = -i\partial_s$ 可写成时间无关哈密顿量
    $H = P_s + H(s)$; 与薛定谔化结合后, 在 $s \otimes p \otimes x$ 三个寄存器上
    \[
        H_h = P_s \otimes I + \sum_{\ell} \ket{\ell}\bra{\ell}
        \otimes \big(D_\mu \otimes H_1(s_\ell) - I \otimes H_2(s_\ell)\big),
    \]
    其中 $D_\mu$ 是 $p$ 方向 Fourier 模态的对角矩阵. 一句话:
    $p$ 变量处理非酉动力学, $s$ 变量处理时间依赖.
\end{Note}

\subsection{复杂度与量子优势}

\begin{Thm}[薛定谔化的复杂度(离散化后)]
    设空间网格 $h$, 维数 $d$, 模拟时间 $T$, 精度 $\varepsilon$.
    Hu--Jin--Liu--Zhang(2024)对显式电路的分析给出:
    \begin{enumerate}
        \item 热方程($u_t = \Delta u$): \textbf{门复杂度(gate complexity)}
        $\widetilde O\!\Big(\frac{d^2 T^2 \norm{u(0)}^3}{h^4 \varepsilon^3 \norm{u(T)}^3}\Big)$,
        当维数 $d > 5$ 时优于经典方法;
        \item 对流方程(迎风): 门复杂度
        $\widetilde O\!\Big(\frac{d T^2 \sum_\alpha c_\alpha^2 \norm{u(0)}^3}{h^2 \varepsilon^3 \norm{u(T)}^3}\Big)$,
        当维数 $d > 4$ 时优于经典方法.
    \end{enumerate}
    经典有限差分在 $d$ 维上的代价随 $d$ 指数增长(\textbf{维度灾难(curse of dimensionality)}),
    而薛定谔化的门复杂度对 $d$ 是多项式 —— 高维问题是\textbf{量子优势(quantum advantage)}的来源.
\end{Thm}

\begin{Note}[与 HHL 路线的对比]
    隐式欧拉 + HHL 路线每时间步调用一次线性方程组求解(共 $M = T/\Delta t$ 次),
    每次都要 QPE 与误差控制; 薛定谔化把整个 $[0,T]$ 上的演化合并为
    \textbf{一次}酉演化, 时间离散的误差被消除, 且不要求空间离散化算子的
    厄米性(迎风格式可直接处理). 两条路线共享 block-encoding 与
    哈密顿量模拟构件, 区别在于"线性化 + 迭代"与"高维\textbf{酉化(unitarization)}"的取舍.
\end{Note}

\begin{Note}[与经典 PDE 数值方法的对比]
    经典求解 $u_t = Lu$ 的主流路线及其代价:
    \begin{itemize}
        \item \textbf{有限差分 / 有限元 / 有限体积(finite difference / finite element / finite volume)}: 网格点数 $N = O(h^{-d})$,
        每时间步 $O(N)$ 次运算; 热方程\textbf{显式格式(explicit scheme)}受 \textbf{CFL 稳定性条件(CFL stability condition)}限制($\Delta t = O(h^2)$),
        \textbf{隐式格式(implicit scheme)}每步需解一个线性方程组($N$ 维);
        \item \textbf{傅里叶谱方法(Fourier spectral method)}: 用 \textbf{FFT(快速傅里叶变换, Fast Fourier Transform)} 把\textbf{微分算子(differential operator)}对角化, 每步 $O(N\log N)$ 次运算,
        对光滑解\textbf{谱精度(spectral accuracy)}指数收敛 —— 但仍受 $N = O(h^{-d})$ 的\textbf{维度灾难}限制:
        $d$ 每增加 1, 计算量乘 $1/h$;
        \item \textbf{自适应 / 多尺度方法}: 缓解但不能根除维度灾难.
    \end{itemize}
    薛定谔化的门复杂度对 $d$ 是多项式($h$ 的幂次与 $d$ 无关的因子乘 $\mathrm{poly}(d)$),
    因此在 $d > 4\text{--}5$ 时对上述经典方法呈指数级优势. 需要强调的前提
    (与 HHL 相同): 初态 $\ket{u_0}$ 可高效制备(稀疏/张量结构)、$H_{\mathrm{eff}}$ 有
    block-encoding、且只提取低维观测 —— 经典方法则始终能读出完整解场.
    这也是"量子加速的诚实边界"在本模块的体现.
\end{Note}

\subsection{病态 PDE: 薛定谔化稳定了什么}

薛定谔化是\textbf{酉化(unitarization)}, 但酉化本身并不自动使病态问题变良态 —— 需要弄清它稳定的是哪一个对象.

\begin{Note}[病态问题为什么难: 反向热方程]
    考虑\textbf{反向热方程(backward heat equation)} $u_t = -u_{xx}$, $x \in \mathbb{T}$.
    其解满足(这里 $\eta$ 是\textbf{空间} Fourier 频率, 区别于辅助变量 $p$ 的
    Fourier 频率 $\xi$; $\hat u_T$ 为终值 $u(T,\cdot)$ 的 Fourier 系数)
    \begin{equation}
        \hat u(t, \eta) = e^{\eta^2 (T-t)}\,\hat u_T(\eta),
    \end{equation}
    高频模态按 $e^{\eta^2(T-t)}$ \textbf{指数增长}: 数据中的微小高频噪声会被指数放大,
    终值问题在 Hadamard 意义下\textbf{不适定(ill-posed)}. 因此不能要求量子算法
    "精确"恢复 $u(t)$ —— 必须在频率截断意义下定义目标解, 并做\textbf{正则化(regularization)}.
\end{Note}

\begin{Note}[薛定谔化稳定的是高维系统, 而非原问题]
    对病态 PDE, 薛定谔化不是简单地把一个已离散好的病态矩阵变成更大的矩阵.
    提升后的 $w(t,x,p)$ 满足\textbf{酉}演化($i\partial_t\hat w = (\xi H_1 + H_2)\hat w$),
    高维系统本身是\textbf{计算稳定}的(模态不指数增长); 但原变量 $u(t,x)$ 的恢复
    仍需要正则化/截断 —— \textbf{薛定谔化并不会让一个任意病态的问题变得良态},
    它只是把"数值不稳定性"与"物理/数学上的不适定性"分离开来: 前者被酉化消除,
    后者由用户通过频率截断显式控制.
\end{Note}

\begin{Note}[PDE-first 与 discretize-first]
    对病态 PDE, 两条构造路线的差异至关重要:
    \begin{enumerate}
        \item \textbf{PDE-first}: 先在连续层面做薛定谔化与频率截断
        ($|\eta| \le \eta_{\max}$, 由可信物理频率决定), 再空间离散:
        \[
            \text{连续 PDE} \;\to\; \text{薛定谔化} \;\to\;
            \text{频率截断/正则化} \;\to\; \text{离散}.
        \]
        正则化误差与离散误差\textbf{分离}, $p$-方向截断、误差分析与量子测量概率
        都只依赖截断后的良态问题;
        \item \textbf{Discretize-first}: 先形成网格矩阵 $A_h$ 再薛定谔化.
        此时恢复区可能被 $\lambda_{\max}(A_h) \sim h^{-2}$ 控制,
        把无意义的网格最高频也纳入辅助截断与成功概率,
        使复杂度被网格最高频"绑架".
    \end{enumerate}
    \textbf{核心要点}: 对病态 PDE, PDE-first 能把正则化误差与离散误差分开,
    discretize-first 容易让网格最高频主导复杂度. 因此应先在连续层面
    定义截断目标, 再离散.
\end{Note}

\begin{Note}[误差分解]
    记频率截断解 $u_{\eta_{\max}} = \mathcal F^{-1}\big[\mathbf 1_{|\eta|\le\eta_{\max}}
    \hat u_T e^{\eta^2(T-t)}\big]$, 总误差可分解为
    \begin{equation}
        \norm{u - u_{h,p}} \;\le\;
        \underbrace{\norm{u - u_{\eta_{\max}}}}_{\text{正则化}}
        + \underbrace{\norm{u_{\eta_{\max}} - u_{\eta_{\max},h}}}_{\text{空间离散}}
        + \underbrace{\norm{w - w_{p,N_p}}}_{p\text{ 截断与谱离散}}
        + \underbrace{\text{量子模拟/测量误差}}_{\text{线路与采样}},
    \end{equation}
    四个来源分别对应 §4.7 的复杂度参数($\eta_{\max}$、$h$、$N_p$ 与 $\varepsilon$),
    需按预算统一分配. 详见 Jin--Liu--Ma 的病态问题分析与参考文献.
\end{Note}

\begin{Note}[非线性 PDE 的展望: Carleman 线性化]
    本模块只处理\textbf{线性}发展方程. 对非线性问题(如半离散 Burgers 方程
    $u_t + uu_x = \nu u_{xx}$: 线性部分耗散、对流项二次), 一个系统化的路线是
    \textbf{Carleman 线性化}(Xiantao Li 讲义 §5.3). 对多项式非线性系统
    \begin{equation}
        \dot u = f_0(t) + F_1(t)u + F_2(t)(u\otimes u),
    \end{equation}
    定义 Carleman 变量 $z_k := u^{\otimes k}$($k \ge 1$, $z_0 = 1$). 对 $z_k$ 求导,
    强迫项降低张量阶、线性项保持张量阶、二次项升高张量阶, 于是整个族满足
    \textbf{分块三对角}的线性系统
    \begin{equation}
        \dot z_k = A_{k,k-1}(t)\, z_{k-1} + A_{k,k}(t)\, z_k + A_{k,k+1}(t)\, z_{k+1},
        \qquad
        A_{k,k}(t) = \sum_{j=1}^{k} I^{\otimes(j-1)}\otimes F_1(t)\otimes I^{\otimes(k-j)},
    \end{equation}
    非线性问题被\textbf{精确}嵌入一个无穷维线性 ODE(代价是无穷多变量);
    截断充当\textbf{闭合(closure)}, 对耗散型系统截断误差可控(收敛率与线性化谱半径有关).
    截断后的线性系统即可套用本模块的 QSVT/薛定谔化构件. 其它路线包括
    \textbf{Koopman--von Neumann 表述}(把密度的平方根看成线性演化对象)与问题相关的
    特化线性化. 需要强调的是: 此类方法的复杂度高度依赖具体非线性结构, 尚无统一最优理论 ——
    非线性量子算法仍以问题特化为主.
\end{Note}

\section{酉化方法族: 薛定谔化、LCHS、Transmutation 与 Dilation}

薛定谔化不是处理非酉动力学唯一的方法. 从"把非酉\textbf{传播子(propagator)}放进更大酉结构"的抽象视角看,
四种常用路线可以放在同一个模板下:
\begin{equation}
    M(T) = e^{TA} \;\approx\;
    \big(\bra{\phi_{\mathrm{out}}}\otimes I\big)\,
    U(T)\,\big(\ket{\phi_{\mathrm{in}}}\otimes I\big),
\end{equation}
其中 $U(T)$ 是更大空间上的酉演化.

\begin{center}
\begin{tabular}{@{}>{\raggedright\arraybackslash}p{4.1cm}>{\raggedright\arraybackslash}p{4.7cm}>{\raggedright\arraybackslash}p{3.0cm}>{\raggedright\arraybackslash}p{2.4cm}@{}}
    \toprule
    方法 & 核心变换 & 最终量子原语 & 最适合的问题 \\
    \midrule
    Schr\"odingerization & 加辅助变量 $p$, $w_t = -H_1 w_p + iH_2 w$ &
    扩大空间上的 Hamiltonian simulation &
    一般线性非酉 ODE/PDE \\
    \textbf{线性组合哈密顿量模拟(Linear Combination of Hamiltonian Simulation, LCHS)} & 对 $w$ 连续 Fourier 后离散 $\xi$ 求积 &
    酉演化的 LCU &
    积分表示清楚的系统 \\
    \textbf{Transmutation(变换法)} & $A = G^\dagger G$, 热半群写成波传播子的 Gaussian 平均 &
    LCU + Hermitian dilation $D(G)$ &
    热方程、扩散型自伴问题 \\
    \textbf{矩匹配膨胀(Moment-matching dilation)} & 构造辅助动力学 $F$, 使 $\bra{\ell}F^q\ket{r} = 1$ &
    大空间 Hamiltonian simulation + projection &
    统一理解各种酉化方法 \\
    \bottomrule
\end{tabular}
\end{center}

\begin{Note}[四种方法的联系]
    \begin{enumerate}
        \item \textbf{LCHS}: 由薛定谔化提升方程先做连续 Fourier 再对 $\xi$ 求积,
        得到 $w(T,p) = \int \hat\psi(\xi)e^{-i\xi p}U_\xi(T)u_0\,d\xi$;
        截断求积后即 LCU. 取 $\psi(p) = e^{-|p|}$ 时
        $\hat\psi(\xi) \propto 1/(1+\xi^2)$, 是 Cauchy 型核.
        \item \textbf{Transmutation}: 利用
        $e^{-Tx^2} = \frac{1}{2\sqrt{\pi T}}\int_{\mathbb{R}} e^{-s^2/(4T)}e^{-isx}\,ds$
        与 $D(G)^2 = \mathrm{diag}(GG^\dagger, G^\dagger G)$(注意 $D(G) = \begin{pmatrix}0&G\\G^\dagger&0\end{pmatrix}$
        的平方把 $A = G^\dagger G$ 放在对角块的相应位置), 把 $e^{-TA}$ 写成
        酉波传播子 $e^{-isD(G)}$ 的 Gaussian 平均; 适合 $A = G^\dagger G$ 的情形.
        \item \textbf{Dilation}: 薛定谔化可看成
        $F = -\partial_p$, $\ket{r} = e^{-p}$, $\bra{\ell} = \delta(p=0)$
        的 moment-matching dilation; 此时 $F^q r = r$, 满足 $\bra{\ell}F^q\ket{r}=1$.
        它把四种方法统一为"稳定非酉流 = 更大空间酉流的投影/平均".
    \end{enumerate}
\end{Note}

\begin{Note}[如何选择]
    一般线性非酉系统优先用薛定谔化, 因为它框架最统一、不要求 $A$ 有特殊结构;
    若已经知道传播子的积分表示, LCHS 的 LCU 线路最自然;
    对正自伴扩散问题, Transmutation 利用一阶因子 $G$
    ($\norm{G} = O(h^{-1})$ 而不是 $\norm{A} = O(h^{-2})$)
    在相干传播代价上常更优;
    若需要比较不同酉化方法或设计新辅助动力学, 用 moment-matching dilation 语言.
\end{Note}
